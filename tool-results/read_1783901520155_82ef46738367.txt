    88→    88→function formatPhone(phone: string): string {
    89→    89→  const digits = phone.replace(/\D/g, '')
    90→    90→  if (digits.length === 11) {
    91→    91→    return digits.replace(/(\d{2})(\d{5})(\d{4})/, '($1) $2-$3')
    92→    92→  }
    93→    93→  if (digits.length === 10) {
    94→    94→    return digits.replace(/(\d{2})(\d{4})(\d{4})/, '($1) $2-$3')
    95→    95→  }
    96→    96→  return phone
    97→    97→}

    99→    99→function showToast(message: string, type: 'success' | 'error' = 'error') {
   100→   100→  // Simple toast using a div
   101→   101→  const existing = document.getElementById('kiteos-toast')
   102→   102→  if (existing) existing.remove()
   103→   103→  const toast = document.createElement('div')
   104→   104→  toast.id = 'kiteos-toast'
   105→   105→  toast.className = `fixed top-4 right-4 z-[9999] px-4 py-3 rounded-lg shadow-lg text-sm font-medium transition-all ${
   106→   106→    type === 'error'
   107→   107→      ? 'bg-red-600 text-white'
   108→   108→      : 'bg-emerald-600 text-white'
   109→   109→  }`
   110→   110→  toast.textContent = message
   111→   111→  document.body.appendChild(toast)
   112→   112→  setTimeout(() => toast.remove(), 4000)
   113→   113→}

   138→   138→function ClientFormDialog({
   139→   139→  open,
   140→   140→  onClose,
   141→   141→  onSave,
   142→   142→  editingClient,
   143→   143→}: {
   144→   144→  open: boolean
   145→   145→  onClose: () => void
   146→   146→  onSave: (data: Record<string, unknown>) => Promise<void>
   147→   147→  editingClient: ClientData | null
   148→   148→}) {
   149→   149→  const [name, setName] = useState('')
   150→   150→  const [cpfCnpj, setCpfCnpj] = useState('')
   151→   151→  const [email, setEmail] = useState('')
   152→   152→  const [phone, setPhone] = useState('')
   153→   153→  const [address, setAddress] = useState('')
   154→   154→  const [city, setCity] = useState('')
   155→   155→  const [state, setState] = useState('')
   156→   156→  const [zipCode, setZipCode] = useState('')
   157→   157→  const [loading, setLoading] = useState(false)
   158→   158→  const [errorMsg, setErrorMsg] = useState('')
   159→   159→  const [lookingUpCNPJ, setLookingUpCNPJ] = useState(false)
   160→   160→  const [lookingUpCEP, setLookingUpCEP] = useState(false)
   161→   161→  const [lookupSuccess, setLookupSuccess] = useState<string | null>(null)
   162→   162→
   163→   163→  useEffect(() => {
   164→   164→    if (editingClient) {
   165→   165→      setName(editingClient.name)
   166→   166→      setCpfCnpj(editingClient.cpfCnpj || '')
   167→   167→      setEmail(editingClient.email || '')
   168→   168→      setPhone(editingClient.phone || '')
   169→   169→      setAddress(editingClient.address || '')
   170→   170→      setCity(editingClient.city || '')
   171→   171→      setState(editingClient.state || '')
   172→   172→      setZipCode(editingClient.zipCode || '')
   173→   173→    } else {
   174→   174→      setName('')
   175→   175→      setCpfCnpj('')
   176→   176→      setEmail('')
   177→   177→      setPhone('')
   178→   178→      setAddress('')
   179→   179→      setCity('')
   180→   180→      setState('')
   181→   181→      setZipCode('')
   182→   182→    }
   183→   183→    setErrorMsg('')
   184→   184→    setLookupSuccess(null)
   185→   185→  }, [editingClient, open])
   186→   186→
   187→   187→  /** Busca automática por CNPJ na BrasilAPI */
   188→   188→  const handleLookupCNPJ = async () => {
   189→   189→    const digits = cpfCnpj.replace(/\D/g, '')
   190→   190→    if (digits.length !== 14) return // Só busca para CNPJ
   191→   191→
   192→   192→    setLookingUpCNPJ(true)
   193→   193→    setLookupSuccess(null)
   194→   194→    try {
   195→   195→      const res = await apiFetch(`/api/consulta/cnpj?cnpj=${digits}`)
   196→   196→      if (res.success && res.data) {
   197→   197→        const d = res.data
   198→   198→        if (d.razaoSocial && !name.trim()) setName(d.razaoSocial)
   199→   199→        if (d.nomeFantasia && !name.trim()) setName(d.nomeFantasia)
   200→   200→        if (d.email && !email.trim()) setEmail(d.email)
   201→   201→        if (d.telefone && !phone.trim()) {
   202→   202→          const phoneDigits = d.telefone.replace(/\D/g, '')
   203→   203→          if (phoneDigits.length >= 10) setPhone(maskPhone(phoneDigits))
   204→   204→        }
   205→   205→        if (d.address && !address.trim()) setAddress(d.address)
   206→   206→        if (d.city && !city.trim()) setCity(d.city)
   207→   207→        if (d.state && !state.trim()) setState(d.state)
   208→   208→        if (d.zipCode && !zipCode.trim()) setZipCode(d.zipCode)
   209→   209→        setLookupSuccess('CNPJ')
   210→   210→      } else if (!res.success) {
   211→   211→        showToast(res.error || 'CNPJ não encontrado', 'error')
   212→   212→      }
   213→   213→    } catch {
   214→   214→      showToast('Erro ao buscar CNPJ', 'error')
   215→   215→    } finally {
   216→   216→      setLookingUpCNPJ(false)
   217→   217→    }
   218→   218→  }
   219→   219→
   220→   220→  /** Busca automática por CEP na BrasilAPI */
   221→   221→  const handleLookupCEP = async () => {
   222→   222→    const digits = zipCode.replace(/\D/g, '')
   223→   223→    if (digits.length !== 8) return
   224→   224→
   225→   225→    setLookingUpCEP(true)
   226→   226→    setLookupSuccess(null)
   227→   227→    try {
   228→   228→      const res = await apiFetch(`/api/consulta/cep?cep=${digits}`)
   229→   229→      if (res.success && res.data) {
   230→   230→        const d = res.data
   231→   231→        if (d.address && !address.trim()) setAddress(d.address)
   232→   232→        if (d.city && !city.trim()) setCity(d.city)
   233→   233→        if (d.state && !state.trim()) setState(d.state)
   234→   234→        setLookupSuccess('CEP')
   235→   235→      } else if (!res.success) {
   236→   236→        showToast(res.error || 'CEP não encontrado', 'error')
   237→   237→      }
   238→   238→    } catch {
   239→   239→      showToast('Erro ao buscar CEP', 'error')
   240→   240→    } finally {
   241→   241→      setLookingUpCEP(false)
   242→   242→    }
   243→   243→  }
   244→   244→
   245→   245→  const handleSave = async () => {
   246→   246→    if (!name.trim()) {
   247→   247→      setErrorMsg('Nome é obrigatório')
   248→   248→      return
   249→   249→    }
   250→   250→    // Validate CPF/CNPJ
   251→   251→    const cpfCnpjError = validateCPFCNPJ(cpfCnpj)
   252→   252→    if (cpfCnpjError) {
   253→   253→      setErrorMsg(cpfCnpjError)
   254→   254→      return
   255→   255→    }
   256→   256→    setErrorMsg('')
   257→   257→    setLoading(true)
   258→   258→    try {
   259→   259→      const payload = {
   260→   260→        name: name.trim(),
   261→   261→        cpfCnpj: cpfCnpj.replace(/\D/g, '') || null,
   262→   262→        email: email.trim() || null,
   263→   263→        phone: phone.replace(/\D/g, '') || null,
   264→   264→        address: address.trim() || null,
   265→   265→        city: city.trim() || null,
   266→   266→        state: state.trim() || null,
   267→   267→        zipCode: zipCode.trim() || null,
   268→   268→      }
   269→   269→
   270→   270→      let res
   271→   271→      if (editingClient) {
   272→   272→        res = await apiFetch(`/api/clients/${editingClient.id}`, {
   273→   273→          method: 'PUT',
   274→   274→          body: JSON.stringify(payload),
   275→   275→        })
   276→   276→      } else {
   277→   277→        const { user } = useAuthStore.getState()
   278→   278→        res = await apiFetch('/api/clients', {
   279→   279→          method: 'POST',
   280→   280→          body: JSON.stringify({ ...payload, officeId: user?.officeId }),
   281→   281→        })
   282→   282→      }
   283→   283→
   284→   284→      if (res.success) {
   285→   285→        showToast(editingClient ? 'Cliente atualizado com sucesso!' : 'Cliente criado com sucesso!', 'success')
   286→   286→        await onSave(payload)
   287→   287→        onClose()
   288→   288→      } else {
   289→   289→        setErrorMsg(res.error || 'Erro ao salvar cliente')
   290→   290→        showToast(res.error || 'Erro ao salvar cliente', 'error')
   291→   291→      }
   292→   292→    } catch (err) {
   293→   293→      console.error('Error saving client:', err)
   294→   294→      const msg = 'Erro de conexão ao salvar cliente'
   295→   295→      setErrorMsg(msg)
   296→   296→      showToast(msg, 'error')
   297→   297→    } finally {
   298→   298→      setLoading(false)
   299→   299→    }
   300→   300→  }
   301→   301→
   302→   302→  return (
   303→   303→    <Dialog open={open} onOpenChange={onClose}>
   304→   304→      <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   305→   305→        <DialogHeader>
   306→   306→          <DialogTitle className="text-lg font-semibold text-zinc-900">
   307→   307→            {editingClient ? 'Editar Cliente' : 'Novo Cliente'}
   308→   308→          </DialogTitle>
   309→   309→        </DialogHeader>
   310→   310→
   311→   311→        <div className="space-y-4">
   312→   312→          {errorMsg && (
   313→   313→            <div className="bg-red-50 border border-red-200 text-red-700 px-4 py-3 rounded-lg text-sm">
   314→   314→              {errorMsg}
   315→   315→            </div>
   316→   316→          )}
   317→   317→
   318→   318→          <div className="space-y-2">
   319→   319→            <Label className="text-sm font-medium text-zinc-700">Nome *</Label>
   320→   320→            <Input
   321→   321→              placeholder="Nome completo"
   322→   322→              value={name}
   323→   323→              onChange={(e) => setName(e.target.value)}
   324→   324→            />
   325→   325→          </div>
   326→   326→
   327→   327→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   328→   328→            <div className="space-y-2">
   329→   329→              <Label className="text-sm font-medium text-zinc-700">CPF/CNPJ</Label>
   330→   330→              <div className="flex gap-2">
   331→   331→                <Input
   332→   332→                  placeholder="000.000.000-00 ou 00.000.000/0000-00"
   333→   333→                  value={cpfCnpj}
   334→   334→                  onChange={(e) => {
   335→   335→                    setCpfCnpj(maskCPFCNPJ(e.target.value))
   336→   336→                    setLookupSuccess(null)
   337→   337→                  }}
   338→   338→                  onBlur={() => {
   339→   339→                    const digits = cpfCnpj.replace(/\D/g, '')
   340→   340→                    if (digits.length === 14) handleLookupCNPJ()
   341→   341→                  }}
   342→   342→                  maxLength={18}
   343→   343→                  className="flex-1"
   344→   344→                />
   345→   345→                {cpfCnpj.replace(/\D/g, '').length === 14 && (
   346→   346→                  <Button
   347→   347→                    type="button"
   348→   348→                    variant="outline"
   349→   349→                    size="sm"
   350→   350→                    onClick={handleLookupCNPJ}
   351→   351→                    disabled={lookingUpCNPJ}
   352→   352→                    className="border-blue-200 text-blue-700 hover:bg-blue-50 gap-1 shrink-0"
   353→   353→                    title="Buscar dados do CNPJ"
   354→   354→                  >
   355→   355→                    {lookingUpCNPJ ? <Loader2 className="w-4 h-4 animate-spin" /> : <SearchCheck className="w-4 h-4" />}
   356→   356→                  </Button>
   357→   357→                )}
   358→   358→              </div>
   359→   359→              {lookupSuccess === 'CNPJ' && (
   360→   360→                <p className="text-xs text-emerald-600 flex items-center gap-1">
   361→   361→                  <SearchCheck className="w-3 h-3" /> Dados preenchidos automaticamente
   362→   362→                </p>
   363→   363→              )}
   364→   364→            </div>
   365→   365→            <div className="space-y-2">
   366→   366→              <Label className="text-sm font-medium text-zinc-700">Telefone</Label>
   367→   367→              <Input
   368→   368→                placeholder="(71) 99999-9999"
   369→   369→                value={phone}
   370→   370→                onChange={(e) => setPhone(maskPhone(e.target.value))}
   371→   371→                maxLength={15}
   372→   372→              />
   373→   373→            </div>
   374→   374→          </div>
   375→   375→
   376→   376→          <div className="space-y-2">
   377→   377→            <Label className="text-sm font-medium text-zinc-700">E-mail</Label>
   378→   378→            <Input

   380→   380→              placeholder="email@exemplo.com"
   381→   381→              value={email}
   382→   382→              onChange={(e) => setEmail(e.target.value)}
   383→   383→            />
   384→   384→          </div>
   385→   385→
   386→   386→          <div className="space-y-2">

   401→   401→                onChange={(e) => setCity(e.target.value)}
   402→   402→              />

   423→   423→                  onBlur={() => {
   424→   424→                    const digits = zipCode.replace(/\D/g, '')
   425→   425→                    if (digits.length === 8 && !address.trim()) handleLookupCEP()
   426→   426→                  }}
   427→   427→                  className="flex-1"
   428→   428→                />
   429→   429→                {zipCode.replace(/\D/g, '').length === 8 && (
   430→   430→                  <Button
   431→   431→                    type="button"
   432→   432→                    variant="outline"
   433→   433→                    size="sm"
   434→   434→                    onClick={handleLookupCEP}
   435→   435→                    disabled={lookingUpCEP}
   436→   436→                    className="border-blue-200 text-blue-700 hover:bg-blue-50 gap-1 shrink-0"
   437→   437→                    title="Buscar endereço pelo CEP"
   438→   438→                  >
   439→   439→                    {lookingUpCEP ? <Loader2 className="w-4 h-4 animate-spin" /> : <SearchCheck className="w-4 h-4" />}
   440→   440→                  </Button>
   441→   441→                )}
   442→   442→              </div>
   443→   443→              {lookupSuccess === 'CEP' && (
   444→   444→                <p className="text-xs text-emerald-600 flex items-center gap-1">
   445→   445→                  <SearchCheck className="w-3 h-3" /> Endereço preenchido pelo CEP

   448→   448→            </div>
   449→   449→          </div>
   450→   450→        </div>
   451→   451→
   452→   452→        <DialogFooter className="gap-2">
   453→   453→          <Button variant="outline" onClick={onClose}>
   454→   454→            Cancelar
   455→   455→          </Button>
   456→   456→          <Button
   457→   457→            onClick={handleSave}
   458→   458→            disabled={loading || !name.trim()}
   459→   459→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   460→   460→          >
   461→   461→            {loading && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
   462→   462→            {editingClient ? 'Salvar Alterações' : 'Criar Cliente'}
   463→   463→          </Button>
   464→   464→        </DialogFooter>
   465→   465→      </DialogContent>
   466→   466→    </Dialog>
   467→   467→  )
