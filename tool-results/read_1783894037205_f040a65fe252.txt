   297→/**
   298→ * Retorna os endpoints CT-e para uma UF e ambiente
   299→ */
   300→function getCteEndpoint(uf: string, ambiente: Ambiente): CteEndpoint {
   301→  const normalizedUf = uf.toUpperCase().trim()
   302→  const endpoints = ambiente === 'PRODUCAO' ? PRODUCAO_CTE_ENDPOINTS : HOMOLOGACAO_CTE_ENDPOINTS
   303→  const endpoint = endpoints[normalizedUf]
   304→  if (!endpoint) {
   305→    return endpoints['RS'] || {
   306→      consulta: 'https://homologacao.cte.sefaz.rs.gov.br/ws/CteConsulta/CteConsulta4.asmx',
   307→      download: 'https://homologacao.cte.sefaz.rs.gov.br/ws/CteDownload/CteDownload4.asmx',
   308→    }
   309→  }
   310→  return endpoint
   311→}

   407→/**
   408→ * Extrai valor de tag XML usando regex
   409→ */
   410→function extractValue(xml: string, tag: string): string | null {
   411→  const cleanTag = tag.replace(/[\w]+:/g, '')
   412→  const patterns = [
   413→    new RegExp(`<[\\w]*:?${escapeRegex(cleanTag)}[^>]*>([^<]+)</[\\w]*:?${escapeRegex(cleanTag)}>`, 'i'),
   414→    new RegExp(`<${escapeRegex(cleanTag)}[^>]*>([^<]+)</${escapeRegex(cleanTag)}>`, 'i'),
   415→  ]
   416→  for (const pattern of patterns) {
   417→    const match = xml.match(pattern)
   418→    if (match && match[1]) return match[1].trim()
   419→  }
   420→  return null
   421→}
   422→
   423→function escapeRegex(str: string): string {
   424→  return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
   425→}
   426→
   427→/**
   428→ * Extrai bloco XML
   429→ */
   430→function extractBlock(xml: string, tag: string): string | null {
   431→  const cleanTag = tag.replace(/[\w]+:/g, '')
   432→  const pattern = new RegExp(
   433→    `<[\\w]*:?${escapeRegex(cleanTag)}[^>]*>([\\s\\S]*?)</[\\w]*:?${escapeRegex(cleanTag)}>`,
   434→    'i'
   435→  )
   436→  const match = xml.match(pattern)
   437→  return match ? match[1] : null
   438→}

   444→function mapCStatToCteStatus(cStat: string): CteConsultaResult['cteStatus'] {
   445→  const authorized = ['100', '150'] // CT-e Autorizado o uso / CT-e Autorizado o uso
   446→  const cancelled = ['101', '151'] // Cancelamento de CT-e / Cancelamento de CT-e fora de prazo

   451→  if (cancelled.includes(cStat)) return 'CANCELADO'

   456→  return 'ERRO'

   459→/**
   460→ * Parse da resposta de Consulta Protocolo CT-e
   461→ */
   462→function parseCteConsultaResponse(soapResponse: string): CteConsultaResult {
   463→  const body = extractBlock(soapResponse, 'soap:Body') || extractBlock(soapResponse, 'Body') || soapResponse
   464→
   465→  const retConsSitCTe = extractBlock(body, 'retConsSitCTe') || body
   466→
   467→  const cStat = extractValue(retConsSitCTe, 'cStat') || ''
   468→  const xMotivo = extractValue(retConsSitCTe, 'xMotivo') || 'Resposta não identificada'
   469→  const chCTe = extractValue(retConsSitCTe, 'chCTe') || ''
   470→  const cUF = extractValue(retConsSitCTe, 'cUF')
   471→
   472→  let protocolo: string | null = null
   473→  let dhRecbto: string | null = null
   474→  let emitenteCnpj: string | null = null
   475→
   476→  const protCTeBlock = extractBlock(retConsSitCTe, 'protCTe')
   477→  if (protCTeBlock) {
   478→    protocolo = extractValue(protCTeBlock, 'nProt')
   479→    dhRecbto = extractValue(protCTeBlock, 'dhRecbto')
   480→
   481→    const infProtBlock = extractBlock(protCTeBlock, 'infProt')
   482→    if (infProtBlock) {
   483→      const emitCnpj = extractValue(infProtBlock, 'CNPJ')
   484→      if (emitCnpj) emitenteCnpj = emitCnpj
   485→    }
   486→  }
   487→
   488→  return {
   489→    cStat,
   490→    xMotivo,
   491→    protocolo,
   492→    dhRecbto,
   493→    chCTe,
   494→    cUF,
   495→    cteStatus: mapCStatToCteStatus(cStat),
   496→    emitenteCnpj,
   497→  }
   498→}

   523→  }
   524→

   766→
   767→  const endpoint = getCteEndpoint(uf, config.ambiente)
   768→  const soapEnvelope = buildCteConsultaProtocoloEnvelope(chave, config.ambiente, uf, config.cnpjCpf)
   769→
   770→  let lastError: string | null = null
   771→  let lastFriendlyError: string | null = null
   772→
   773→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
   774→    try {
   775→      const controller = new AbortController()
   776→      const timeoutId = setTimeout(() => controller.abort(), timeout)
   777→
   778→      const response = await fetch(endpoint.consulta, {
   779→        method: 'POST',
   780→        headers: {
   781→          'Content-Type': 'application/soap+xml; charset=utf-8',
   782→          'SOAPAction': 'http://www.portalfiscal.inf.br/cte/wsdl/CteConsultaProtocolo4/cteConsultaCT',
   783→        },
   784→        body: soapEnvelope,
   785→        signal: controller.signal,
   786→      })
   787→
   788→      clearTimeout(timeoutId)
   789→
   790→      if (!response.ok) {
   791→        lastError = `HTTP ${response.status}: ${response.statusText}`
   792→        lastFriendlyError = response.status === 503
   793→          ? 'SEFAZ indisponível no momento. Tente novamente em alguns minutos'
   794→          : response.status === 403
   795→            ? 'Certificado digital não autorizado para este webservice'
   796→            : `Erro na comunicação com SEFAZ CT-e (HTTP ${response.status})`
   797→
   798→        if (attempt < maxRetries) {
   799→          await delay(RETRY_DELAY_MS * attempt)
   800→          continue
   801→        }
   802→        break
   803→      }
   804→
   805→      const responseBody = await response.text()
   806→      const result = parseCteConsultaResponse(responseBody)
   807→      const tempoResposta = Date.now() - startTime
   808→
   809→      if (result.cStat && result.cteStatus === 'ERRO' && !isRetryableError(result.cStat)) {
   810→        return {
   811→          success: false,
   812→          data: result,
   813→          error: `${result.cStat} - ${result.xMotivo}`,
   814→          friendlyError: mapCteErrorToFriendly(result.cStat, result.xMotivo),
   815→          tempoResposta,
   816→          tentativas: attempt,
   817→        }
   818→      }
   819→
   820→      return {
   821→        success: result.cteStatus === 'AUTORIZADO',
   822→        data: result,
   823→        tempoResposta,
   824→        tentativas: attempt,
   825→      }
   826→    } catch (err) {
   827→      const isTimeout = err instanceof Error && err.name === 'AbortError'
   828→      lastError = isTimeout
   829→        ? `Timeout após ${timeout}ms`
   830→        : err instanceof Error ? err.message : 'Erro desconhecido'
   831→      lastFriendlyError = isTimeout
   832→        ? 'Tempo de resposta da SEFAZ excedido. Tente novamente'
   833→        : 'Erro de comunicação com a SEFAZ CT-e. Verifique sua conexão'
   834→
   835→      if (attempt < maxRetries) {
   836→        await delay(RETRY_DELAY_MS * attempt)
   837→        continue
   838→      }
   839→    }
   840→  }
   841→
   842→  const tempoResposta = Date.now() - startTime
   843→  return {
   844→    success: false,
   845→    error: lastError || 'Erro desconhecido',
   846→    friendlyError: lastFriendlyError || 'Erro na comunicação com SEFAZ CT-e',
   847→    tempoResposta,
   848→    tentativas: maxRetries,
   849→  }
   850→}
