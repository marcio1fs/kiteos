    88→function formatPhone(phone: string): string {
    89→  const digits = phone.replace(/\D/g, '')
    90→  if (digits.length === 11) {
    91→    return digits.replace(/(\d{2})(\d{5})(\d{4})/, '($1) $2-$3')
    92→  }
    93→  if (digits.length === 10) {
    94→    return digits.replace(/(\d{2})(\d{4})(\d{4})/, '($1) $2-$3')
    95→  }
    96→  return phone
    97→}

    99→function showToast(message: string, type: 'success' | 'error' = 'error') {
   100→  // Simple toast using a div
   101→  const existing = document.getElementById('kiteos-toast')
   102→  if (existing) existing.remove()
   103→  const toast = document.createElement('div')
   104→  toast.id = 'kiteos-toast'
   105→  toast.className = `fixed top-4 right-4 z-[9999] px-4 py-3 rounded-lg shadow-lg text-sm font-medium transition-all ${
   106→    type === 'error'
   107→      ? 'bg-red-600 text-white'
   108→      : 'bg-emerald-600 text-white'
   109→  }`
   110→  toast.textContent = message
   111→  document.body.appendChild(toast)
   112→  setTimeout(() => toast.remove(), 4000)
   113→}

   138→function ClientFormDialog({
   139→  open,
   140→  onClose,
   141→  onSave,
   142→  editingClient,
   143→}: {
   144→  open: boolean
   145→  onClose: () => void
   146→  onSave: (data: Record<string, unknown>) => Promise<void>
   147→  editingClient: ClientData | null
   148→}) {
   149→  const [name, setName] = useState('')
   150→  const [cpfCnpj, setCpfCnpj] = useState('')
   151→  const [email, setEmail] = useState('')
   152→  const [phone, setPhone] = useState('')
   153→  const [address, setAddress] = useState('')
   154→  const [city, setCity] = useState('')
   155→  const [state, setState] = useState('')
   156→  const [zipCode, setZipCode] = useState('')
   157→  const [loading, setLoading] = useState(false)
   158→  const [errorMsg, setErrorMsg] = useState('')
   159→  const [lookingUpCNPJ, setLookingUpCNPJ] = useState(false)
   160→  const [lookingUpCEP, setLookingUpCEP] = useState(false)
   161→  const [lookupSuccess, setLookupSuccess] = useState<string | null>(null)
   162→
   163→  useEffect(() => {
   164→    if (editingClient) {
   165→      setName(editingClient.name)
   166→      setCpfCnpj(editingClient.cpfCnpj || '')
   167→      setEmail(editingClient.email || '')
   168→      setPhone(editingClient.phone || '')
   169→      setAddress(editingClient.address || '')
   170→      setCity(editingClient.city || '')
   171→      setState(editingClient.state || '')
   172→      setZipCode(editingClient.zipCode || '')
   173→    } else {
   174→      setName('')
   175→      setCpfCnpj('')
   176→      setEmail('')
   177→      setPhone('')
   178→      setAddress('')
   179→      setCity('')
   180→      setState('')
   181→      setZipCode('')
   182→    }
   183→    setErrorMsg('')
   184→    setLookupSuccess(null)
   185→  }, [editingClient, open])
   186→
   187→  /** Busca automática por CNPJ na BrasilAPI */
   188→  const handleLookupCNPJ = async () => {
   189→    const digits = cpfCnpj.replace(/\D/g, '')
   190→    if (digits.length !== 14) return // Só busca para CNPJ
   191→
   192→    setLookingUpCNPJ(true)
   193→    setLookupSuccess(null)
   194→    try {
   195→      const res = await apiFetch(`/api/consulta/cnpj?cnpj=${digits}`)
   196→      if (res.success && res.data) {
   197→        const d = res.data
   198→        if (d.razaoSocial && !name.trim()) setName(d.razaoSocial)
   199→        if (d.nomeFantasia && !name.trim()) setName(d.nomeFantasia)
   200→        if (d.email && !email.trim()) setEmail(d.email)
   201→        if (d.telefone && !phone.trim()) {
   202→          const phoneDigits = d.telefone.replace(/\D/g, '')
   203→          if (phoneDigits.length >= 10) setPhone(maskPhone(phoneDigits))
   204→        }
   205→        if (d.address && !address.trim()) setAddress(d.address)
   206→        if (d.city && !city.trim()) setCity(d.city)
   207→        if (d.state && !state.trim()) setState(d.state)
   208→        if (d.zipCode && !zipCode.trim()) setZipCode(d.zipCode)
   209→        setLookupSuccess('CNPJ')
   210→      } else if (!res.success) {
   211→        showToast(res.error || 'CNPJ não encontrado', 'error')
   212→      }
   213→    } catch {
   214→      showToast('Erro ao buscar CNPJ', 'error')
   215→    } finally {
   216→      setLookingUpCNPJ(false)
   217→    }
   218→  }
   219→
   220→  /** Busca automática por CEP na BrasilAPI */
   221→  const handleLookupCEP = async () => {
   222→    const digits = zipCode.replace(/\D/g, '')
   223→    if (digits.length !== 8) return
   224→
   225→    setLookingUpCEP(true)
   226→    setLookupSuccess(null)
   227→    try {
   228→      const res = await apiFetch(`/api/consulta/cep?cep=${digits}`)
   229→      if (res.success && res.data) {
   230→        const d = res.data
   231→        if (d.address && !address.trim()) setAddress(d.address)
   232→        if (d.city && !city.trim()) setCity(d.city)
   233→        if (d.state && !state.trim()) setState(d.state)
   234→        setLookupSuccess('CEP')
   235→      } else if (!res.success) {
   236→        showToast(res.error || 'CEP não encontrado', 'error')
   237→      }
   238→    } catch {
   239→      showToast('Erro ao buscar CEP', 'error')
   240→    } finally {
   241→      setLookingUpCEP(false)
   242→    }
   243→  }
   244→
   245→  const handleSave = async () => {
   246→    if (!name.trim()) {
   247→      setErrorMsg('Nome é obrigatório')
   248→      return
   249→    }
   250→    // Validate CPF/CNPJ
   251→    const cpfCnpjError = validateCPFCNPJ(cpfCnpj)
   252→    if (cpfCnpjError) {
   253→      setErrorMsg(cpfCnpjError)
   254→      return
   255→    }
   256→    setErrorMsg('')
   257→    setLoading(true)
   258→    try {
   259→      const payload = {
   260→        name: name.trim(),
   261→        cpfCnpj: cpfCnpj.replace(/\D/g, '') || null,
   262→        email: email.trim() || null,
   263→        phone: phone.replace(/\D/g, '') || null,
   264→        address: address.trim() || null,
   265→        city: city.trim() || null,
   266→        state: state.trim() || null,
   267→        zipCode: zipCode.trim() || null,
   268→      }
   269→
   270→      let res
   271→      if (editingClient) {
   272→        res = await apiFetch(`/api/clients/${editingClient.id}`, {
   273→          method: 'PUT',
   274→          body: JSON.stringify(payload),
   275→        })
   276→      } else {
   277→        const { user } = useAuthStore.getState()
   278→        res = await apiFetch('/api/clients', {
   279→          method: 'POST',
   280→          body: JSON.stringify({ ...payload, officeId: user?.officeId }),
   281→        })
   282→      }
   283→
   284→      if (res.success) {
   285→        showToast(editingClient ? 'Cliente atualizado com sucesso!' : 'Cliente criado com sucesso!', 'success')
   286→        await onSave(payload)
   287→        onClose()
   288→      } else {
   289→        setErrorMsg(res.error || 'Erro ao salvar cliente')
   290→        showToast(res.error || 'Erro ao salvar cliente', 'error')
   291→      }
   292→    } catch (err) {
   293→      console.error('Error saving client:', err)
   294→      const msg = 'Erro de conexão ao salvar cliente'
   295→      setErrorMsg(msg)
   296→      showToast(msg, 'error')
   297→    } finally {
   298→      setLoading(false)
   299→    }
   300→  }
   301→
   302→  return (
   303→    <Dialog open={open} onOpenChange={onClose}>
   304→      <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   305→        <DialogHeader>
   306→          <DialogTitle className="text-lg font-semibold text-zinc-900">
   307→            {editingClient ? 'Editar Cliente' : 'Novo Cliente'}
   308→          </DialogTitle>
   309→        </DialogHeader>
   310→
   311→        <div className="space-y-4">
   312→          {errorMsg && (
   313→            <div className="bg-red-50 border border-red-200 text-red-700 px-4 py-3 rounded-lg text-sm">
   314→              {errorMsg}
   315→            </div>
   316→          )}
   317→
   318→          <div className="space-y-2">
   319→            <Label className="text-sm font-medium text-zinc-700">Nome *</Label>
   320→            <Input
   321→              placeholder="Nome completo"
   322→              value={name}
   323→              onChange={(e) => setName(e.target.value)}
   324→            />
   325→          </div>
   326→
   327→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   328→            <div className="space-y-2">
   329→              <Label className="text-sm font-medium text-zinc-700">CPF/CNPJ</Label>
   330→              <div className="flex gap-2">
   331→                <Input
   332→                  placeholder="000.000.000-00 ou 00.000.000/0000-00"
   333→                  value={cpfCnpj}
   334→                  onChange={(e) => {
   335→                    setCpfCnpj(maskCPFCNPJ(e.target.value))
   336→                    setLookupSuccess(null)
   337→                  }}
   338→                  onBlur={() => {
   339→                    const digits = cpfCnpj.replace(/\D/g, '')
   340→                    if (digits.length === 14) handleLookupCNPJ()
   341→                  }}
   342→                  maxLength={18}
   343→                  className="flex-1"
   344→                />
   345→                {cpfCnpj.replace(/\D/g, '').length === 14 && (
   346→                  <Button
   347→                    type="button"
   348→                    variant="outline"
   349→                    size="sm"
   350→                    onClick={handleLookupCNPJ}
   351→                    disabled={lookingUpCNPJ}
   352→                    className="border-blue-200 text-blue-700 hover:bg-blue-50 gap-1 shrink-0"
   353→                    title="Buscar dados do CNPJ"
   354→                  >
   355→                    {lookingUpCNPJ ? <Loader2 className="w-4 h-4 animate-spin" /> : <SearchCheck className="w-4 h-4" />}
   356→                  </Button>
   357→                )}
   358→              </div>
   359→              {lookupSuccess === 'CNPJ' && (
   360→                <p className="text-xs text-emerald-600 flex items-center gap-1">
   361→                  <SearchCheck className="w-3 h-3" /> Dados preenchidos automaticamente
   362→                </p>
   363→              )}
   364→            </div>
   365→            <div className="space-y-2">
   366→              <Label className="text-sm font-medium text-zinc-700">Telefone</Label>
   367→              <Input
   368→                placeholder="(71) 99999-9999"
   369→                value={phone}
   370→                onChange={(e) => setPhone(maskPhone(e.target.value))}
   371→                maxLength={15}
   372→              />
   373→            </div>
   374→          </div>
   375→
   376→          <div className="space-y-2">
   377→            <Label className="text-sm font-medium text-zinc-700">E-mail</Label>
   378→            <Input
   379→              type="email"
   380→              placeholder="email@exemplo.com"
   381→              value={email}
   382→              onChange={(e) => setEmail(e.target.value)}
   383→            />
   384→          </div>
   385→
   386→          <div className="space-y-2">
   387→            <Label className="text-sm font-medium text-zinc-700">Endereço</Label>
   388→            <Input
   389→              placeholder="Rua, número, complemento"
   390→              value={address}
   391→              onChange={(e) => setAddress(e.target.value)}
   392→            />
   393→          </div>
   394→
   395→          <div className="grid grid-cols-3 gap-4">
   396→            <div className="space-y-2">
   397→              <Label className="text-sm font-medium text-zinc-700">Cidade</Label>
   398→              <Input
   399→                placeholder="Cidade"
   400→                value={city}
   401→                onChange={(e) => setCity(e.target.value)}
   402→              />
   403→            </div>
   404→            <div className="space-y-2">
   405→              <Label className="text-sm font-medium text-zinc-700">Estado</Label>
   406→              <Input
   407→                placeholder="UF"
   408→                maxLength={2}
   409→                value={state}
   410→                onChange={(e) => setState(e.target.value.toUpperCase())}
   411→              />
   412→            </div>
   413→            <div className="space-y-2">
   414→              <Label className="text-sm font-medium text-zinc-700">CEP</Label>
   415→              <div className="flex gap-2">
   416→                <Input
   417→                  placeholder="00000-000"
   418→                  value={zipCode}
   419→                  onChange={(e) => {
   420→                    setZipCode(e.target.value)
   421→                    if (lookupSuccess === 'CEP') setLookupSuccess(null)
   422→                  }}
   423→                  onBlur={() => {
   424→                    const digits = zipCode.replace(/\D/g, '')
   425→                    if (digits.length === 8 && !address.trim()) handleLookupCEP()
   426→                  }}
   427→                  className="flex-1"
   428→                />
   429→                {zipCode.replace(/\D/g, '').length === 8 && (
   430→                  <Button
   431→                    type="button"
   432→                    variant="outline"
   433→                    size="sm"
   434→                    onClick={handleLookupCEP}
   435→                    disabled={lookingUpCEP}
   436→                    className="border-blue-200 text-blue-700 hover:bg-blue-50 gap-1 shrink-0"
   437→                    title="Buscar endereço pelo CEP"
   438→                  >
   439→                    {lookingUpCEP ? <Loader2 className="w-4 h-4 animate-spin" /> : <SearchCheck className="w-4 h-4" />}
   440→                  </Button>
   441→                )}
   442→              </div>
   443→              {lookupSuccess === 'CEP' && (
   444→                <p className="text-xs text-emerald-600 flex items-center gap-1">
   445→                  <SearchCheck className="w-3 h-3" /> Endereço preenchido pelo CEP
   446→                </p>
   447→              )}
   448→            </div>
   449→          </div>
   450→        </div>
   451→
   452→        <DialogFooter className="gap-2">
   453→          <Button variant="outline" onClick={onClose}>
   454→            Cancelar
   455→          </Button>
   456→          <Button
   457→            onClick={handleSave}
   458→            disabled={loading || !name.trim()}
   459→            className="bg-emerald-600 hover:bg-emerald-700 text-white"
   460→          >
   461→            {loading && <Loader2 className="w-4 h-4 mr-2 animate-spin" />}
   462→            {editingClient ? 'Salvar Alterações' : 'Criar Cliente'}
   463→          </Button>
   464→        </DialogFooter>
   465→      </DialogContent>
   466→    </Dialog>
   467→  )
   468→}

   473→function VehicleFormDialog({
   474→  open,
   475→  onClose,
   476→  onSave,
   477→  clientId,
   478→  editingVehicle,
   479→}: {
   480→  open: boolean
   481→  onClose: () => void
   482→  onSave: (data: Record<string, unknown>) => Promise<void>
   483→  clientId: string
   484→  editingVehicle: VehicleData | null
   485→}) {
   486→  const { user } = useAuthStore()
   487→  const [plate, setPlate] = useState('')
   488→  const [brand, setBrand] = useState('')
   489→  const [model, setModel] = useState('')
   490→  const [year, setYear] = useState('')
   491→  const [color, setColor] = useState('')
   492→  const [km, setKm] = useState('')
   493→  const [chassi, setChassi] = useState('')
   494→  const [loading, setLoading] = useState(false)
   495→  const [errorMsg, setErrorMsg] = useState('')
   496→  const [lookingUpPlaca, setLookingUpPlaca] = useState(false)
   497→  const [lookupSuccess, setLookupSuccess] = useState(false)
   498→
   499→  useEffect(() => {
   500→    if (editingVehicle) {
   501→      setPlate(editingVehicle.plate)
   502→      setBrand(editingVehicle.brand || '')
   503→      setModel(editingVehicle.model || '')
   504→      setYear(editingVehicle.year?.toString() || '')
   505→      setColor(editingVehicle.color || '')
   506→      setKm(editingVehicle.km?.toString() || '')
   507→      setChassi(editingVehicle.chassi || '')
   508→    } else {
   509→      setPlate('')
   510→      setBrand('')
   511→      setModel('')
   512→      setYear('')
   513→      setColor('')
   514→      setKm('')
   515→      setChassi('')
   516→    }
   517→    setErrorMsg('')
   518→    setLookupSuccess(false)
   519→  }, [editingVehicle, open])
   520→
   521→  /** Busca automática por Placa na BrasilAPI */
   522→  const handleLookupPlaca = async () => {
   523→    const cleanPlate = plate.toUpperCase().replace(/[^A-Z0-9]/g, '')
   524→    if (!cleanPlate || (cleanPlate.length !== 7 && cleanPlate.length !== 8)) return
   525→
   526→    setLookingUpPlaca(true)
   527→    setLookupSuccess(false)
   528→    try {
   529→      const res = await apiFetch(`/api/consulta/placa?placa=${cleanPlate}`)
   530→      if (res.success && res.data) {
   531→        const d = res.data
   532→        if (d.brand && !brand.trim()) setBrand(d.brand)
   533→        if (d.model && !model.trim()) setModel(d.model)
   534→        if (d.year && !year.trim()) setYear(String(d.year))
   535→        if (d.color && !color.trim()) setColor(d.color)
   536→        if (d.chassi && !chassi.trim()) setChassi(d.chassi)
   537→        setLookupSuccess(true)
   538→      } else if (!res.success) {
   539→        showToast(res.error || 'Placa não encontrada', 'error')
   540→      }
   541→    } catch {
   542→      showToast('Erro ao buscar placa', 'error')
   543→    } finally {
   544→      setLookingUpPlaca(false)
   545→    }
   546→  }
   547→
   548→  const handleSave = async () => {
   549→    if (!plate.trim()) {
   550→      setErrorMsg('Placa é obrigatória')
   551→      return
   552→    }
   553→    if (!clientId) {
   554→      setErrorMsg('Selecione um cliente para vincular o veículo')
   555→      return
   556→    }
   557→    setErrorMsg('')
   558→    setLoading(true)
   559→    try {
   560→      const payload = {
   561→        plate: plate.toUpperCase().trim(),
   562→        brand: brand.trim() || null,
   563→        model: model.trim() || null,
   564→        year: year ? parseInt(year) : null,
   565→        color: color.trim() || null,
   566→        km: km ? parseInt(km) : null,
   567→        chassi: chassi.trim() || null,
   568→        clientId,
   569→        officeId: user?.officeId,
   570→      }
   571→
   572→      let res
   573→      if (editingVehicle) {
   574→        res = await apiFetch(`/api/vehicles/${editingVehicle.id}`, {
