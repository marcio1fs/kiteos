    14→  xmlContent: string | null  // XML da NF-e (se disponível na resposta)
    15→  fornecedorCnpj: string | null  // CNPJ do emitente

    25→/**
    26→ * Extrai valor de tag XML usando regex (mesmo padrão do xml-parser)
    27→ */
    28→function extractValue(xml: string, tag: string): string | null {
    29→  const cleanTag = tag.replace(/[\w]+:/g, '')
    30→  const patterns = [
    31→    new RegExp(`<[\\w]*:?${escapeRegex(cleanTag)}[^>]*>([^<]+)</[\\w]*:?${escapeRegex(cleanTag)}>`, 'i'),
    32→    new RegExp(`<${escapeRegex(cleanTag)}[^>]*>([^<]+)</${escapeRegex(cleanTag)}>`, 'i'),
    33→  ]
    34→  for (const pattern of patterns) {
    35→    const match = xml.match(pattern)
    36→    if (match && match[1]) return match[1].trim()
    37→  }
    38→  return null
    39→}
    40→
    41→function escapeRegex(str: string): string {
    42→  return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    43→}
    44→
    45→/**
    46→ * Extrai bloco XML
    47→ */
    48→function extractBlock(xml: string, tag: string): string | null {
    49→  const cleanTag = tag.replace(/[\w]+:/g, '')
    50→  const pattern = new RegExp(
    51→    `<[\\w]*:?${escapeRegex(cleanTag)}[^>]*>([\\s\\S]*?)</[\\w]*:?${escapeRegex(cleanTag)}>`,
    52→    'i'
    53→  )
    54→  const match = xml.match(pattern)
    55→  return match ? match[1] : null
    56→}

    58→/**
    59→ * Mapeia cStat para status interno
    60→ */
    61→function mapCStatToStatus(cStat: string): SefazConsultaResult['nfeStatus'] {
    62→  const authorized = ['100', '150'] // Autorizado o uso da NF-e / Autorizado o uso da NF-e
    63→  const cancelled = ['101', '151', '218'] // Cancelamento / Cancelamento por substituição
    64→  const denied = ['110', '205', '301', '302'] // Uso Denegado
    65→  const unused = ['102', '217', '562'] // Inutilização
    66→
    67→  if (authorized.includes(cStat)) return 'AUTORIZADA'
    68→  if (cancelled.includes(cStat)) return 'CANCELADA'
    69→  if (denied.includes(cStat)) return 'DENEGADA'
    70→  if (unused.includes(cStat)) return 'INUTILIZADA'
    71→  if (['516', '517', '518', '563', '564', '565', '567', '568', '578', '579', '580'].includes(cStat)) return 'NAO_ENCONTRADA'
    72→
    73→  return 'ERRO'
    74→}

    76→/**
    77→ * Parse da resposta de Consulta Protocolo
    78→ * Extrai dados do envelope SOAP retornado pela SEFAZ
    79→ */
    80→export function parseConsultaProtocoloResponse(soapResponse: string): SefazConsultaResult {
    81→  // Extrair corpo do SOAP
    82→  const body = extractBlock(soapResponse, 'soap:Body') || extractBlock(soapResponse, 'Body') || soapResponse
    83→
    84→  // Extrair retConsSitNFe
    85→  const retConsSitNFe = extractBlock(body, 'retConsSitNFe') || body
    86→
    87→  const cStat = extractValue(retConsSitNFe, 'cStat') || ''
    88→  const xMotivo = extractValue(retConsSitNFe, 'xMotivo') || 'Resposta não identificada'
    89→  const chNFe = extractValue(retConsSitNFe, 'chNFe') || ''
    90→  const cUF = extractValue(retConsSitNFe, 'cUF')
    91→
    92→  // Extrair protocolo de autorização
    93→  let protocolo: string | null = null
    94→  let dhRecbto: string | null = null
    95→  let fornecedorCnpj: string | null = null
    96→
    97→  // Buscar protocolo dentro de protNFe
    98→  const protNFeBlock = extractBlock(retConsSitNFe, 'protNFe')
    99→  if (protNFeBlock) {
   100→    protocolo = extractValue(protNFeBlock, 'nProt')
   101→    dhRecbto = extractValue(protNFeBlock, 'dhRecbto')
   102→
   103→    // Tentar extrair CNPJ do emitente do XML do protocolo
   104→    const infProtBlock = extractBlock(protNFeBlock, 'infProt')
   105→    if (infProtBlock) {
   106→      // Se houver XML dentro do protocolo, extrair emit/CNPJ
   107→      const emitCnpj = extractValue(infProtBlock, 'CNPJ')
   108→      if (emitCnpj) fornecedorCnpj = emitCnpj
   109→    }
   110→  }
   111→
   112→  return {
   113→    cStat,
   114→    xMotivo,
   115→    protocolo,
   116→    dhRecbto,
   117→    chNFe,
   118→    cUF,
   119→    nfeStatus: mapCStatToStatus(cStat),
   120→    xmlContent: null, // Consulta protocolo não retorna XML completo
   121→    fornecedorCnpj,
   122→  }
   123→}

   125→/**
   126→ * Parse da resposta de Download NF-e
   127→ * Extrai XML autorizado da resposta
   128→ */
   129→export function parseDownloadNfeResponse(soapResponse: string): SefazDownloadResult {
   130→  const body = extractBlock(soapResponse, 'soap:Body') || extractBlock(soapResponse, 'Body') || soapResponse
   131→
   132→  const retDownloadNFe = extractBlock(body, 'retDownloadNFe') || body
   133→
   134→  const cStat = extractValue(retDownloadNFe, 'cStat') || ''
   135→  const xMotivo = extractValue(retDownloadNFe, 'xMotivo') || 'Resposta não identificada'
   136→  const chNFe = extractValue(retDownloadNFe, 'chNFe') || ''
   137→
   138→  // Extrair XML da NF-e
   139→  let xmlContent: string | null = null
   140→  const procNFeBlock = extractBlock(retDownloadNFe, 'procNFe')
   141→  if (procNFeBlock) {
   142→    // O procNFe contém o NFe + protNFe completo
   143→    // Precisamos reconstruir o XML com a tag raiz
   144→    xmlContent = `<?xml version="1.0" encoding="UTF-8"?><nfeProc xmlns="http://www.portalfiscal.inf.br/nfe">${procNFeBlock}</nfeProc>`
   145→  }
   146→
   147→  // Se não encontrou procNFe, tentar extrair NFe diretamente
   148→  if (!xmlContent) {
   149→    const nfeBlock = extractBlock(retDownloadNFe, 'NFe')
   150→    if (nfeBlock) {
   151→      xmlContent = `<?xml version="1.0" encoding="UTF-8"?><NFe xmlns="http://www.portalfiscal.inf.br/nfe">${nfeBlock}</NFe>`
   152→    }
   153→  }
   154→
   155→  return {
   156→    cStat,
   157→    xMotivo,
   158→    chNFe,
   159→    xmlContent,
   160→  }
   161→}

   164→ * Parse da resposta de Status Serviço

   166→export function parseStatusServicoResponse(soapResponse: string): {
   167→  cStat: string
   168→  xMotivo: string
   169→  dhRecbto: string | null
   170→  tMed: string | null
   171→  dhRetorno: string | null
   172→  xObs: string | null
   173→} {
   174→  const body = extractBlock(soapResponse, 'soap:Body') || extractBlock(soapResponse, 'Body') || soapResponse
   175→  const retConsStatServ = extractBlock(body, 'retConsStatServ') || body
   176→
   177→  return {
   178→    cStat: extractValue(retConsStatServ, 'cStat') || '',
   179→    xMotivo: extractValue(retConsStatServ, 'xMotivo') || '',
   180→    dhRecbto: extractValue(retConsStatServ, 'dhRecbto'),
   181→    tMed: extractValue(retConsStatServ, 'tMed'),
   182→    dhRetorno: extractValue(retConsStatServ, 'dhRetorno'),
   183→    xObs: extractValue(retConsStatServ, 'xObs'),
   184→  }
   185→}
