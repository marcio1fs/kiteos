   134→const STATUS_CONFIG: Record<OSStatus, { label: string; color: string; icon: typeof FileText }> = {
   135→  ABERTA: { label: 'Aberta', color: 'bg-blue-100 text-blue-800 border-blue-200', icon: FileText },
   136→  EM_EXECUCAO: { label: 'Em Execução', color: 'bg-yellow-100 text-yellow-800 border-yellow-200', icon: Play },
   137→  FINALIZADA: { label: 'Finalizada', color: 'bg-green-100 text-green-800 border-green-200', icon: CheckCircle },
   138→  PAGA: { label: 'Paga', color: 'bg-emerald-100 text-emerald-800 border-emerald-200', icon: CreditCard },
   139→  CANCELADA: { label: 'Cancelada', color: 'bg-red-100 text-red-800 border-red-200', icon: XCircle },
   140→}
   141→
   142→const STATUS_FLOW: OSStatus[] = ['ABERTA', 'EM_EXECUCAO', 'FINALIZADA', 'PAGA']
   143→
   144→function StatusBadge({ status }: { status: OSStatus }) {
   145→  const config = STATUS_CONFIG[status]
   146→  return (
   147→    <Badge variant="outline" className={`${config.color} font-medium`}>
   148→      {config.label}
   149→    </Badge>
   150→  )
   151→}
   152→
   153→function formatCurrency(value: number) {
   154→  return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(value)
   155→}
   156→
   157→function formatDate(dateStr: string) {
   158→  return new Date(dateStr).toLocaleDateString('pt-BR', {
   159→    day: '2-digit',
   160→    month: '2-digit',
   161→    year: 'numeric',
   162→  })
   163→}

   165→// =============================================
   166→// STATUS TIMELINE
   167→// =============================================
   168→function StatusTimeline({ status }: { status: OSStatus }) {
   169→  const currentIndex = STATUS_FLOW.indexOf(status)
   170→  const isCancelled = status === 'CANCELADA'
   171→
   172→  return (
   173→    <div className="flex items-center justify-between w-full">
   174→      {STATUS_FLOW.map((step, index) => {
   175→        const config = STATUS_CONFIG[step]
   176→        const isCompleted = !isCancelled && index <= currentIndex
   177→        const isCurrent = !isCancelled && index === currentIndex
   178→        const Icon = config.icon
   179→
   180→        return (
   181→          <div key={step} className="flex items-center flex-1 last:flex-none">
   182→            <div className="flex flex-col items-center gap-1.5">
   183→              <div
   184→                className={`w-10 h-10 rounded-full flex items-center justify-center border-2 transition-all ${
   185→                  isCompleted
   186→                    ? 'bg-emerald-600 border-emerald-600 text-white'
   187→                    : isCancelled
   188→                    ? 'bg-zinc-100 border-zinc-300 text-zinc-400'
   189→                    : 'bg-white border-zinc-300 text-zinc-400'
   190→                } ${isCurrent ? 'ring-4 ring-emerald-100' : ''}`}
   191→              >
   192→                <Icon className="w-4 h-4" />
   193→              </div>
   194→              <span
   195→                className={`text-xs font-medium ${
   196→                  isCompleted ? 'text-emerald-700' : isCancelled ? 'text-zinc-400' : 'text-zinc-400'
   197→                }`}
   198→              >
   199→                {config.label}
   200→              </span>
   201→            </div>
   202→            {index < STATUS_FLOW.length - 1 && (
   203→              <div
   204→                className={`flex-1 h-0.5 mx-2 rounded-full transition-all ${
   205→                  !isCancelled && index < currentIndex ? 'bg-emerald-600' : 'bg-zinc-200'
   206→                }`}
   207→              />
   208→            )}
   209→          </div>
   210→        )
   211→      })}
   212→    </div>
   213→  )
   214→}

   216→// =============================================
   217→// NEW/EDIT OS DIALOG
   218→// =============================================
   219→function OSFormDialog({
   220→  open,
   221→  onClose,
   222→  onSave,
   223→  editingOS,
   224→}: {
   225→  open: boolean
   226→  onClose: () => void
   227→  onSave: (data: Record<string, unknown>) => Promise<void>
   228→  editingOS: ServiceOrder | null
   229→}) {
   230→  const { user } = useAuthStore()
   231→  const officeId = user?.officeId || ''
   232→
   233→  const [clients, setClients] = useState<ClientSimple[]>([])
   234→  const [vehicles, setVehicles] = useState<VehicleSimple[]>([])
   235→  const [parts, setParts] = useState<PartSimple[]>([])
   236→  const [mechanics, setMechanics] = useState<MechanicSimple[]>([])
   237→  const [loading, setLoading] = useState(false)
   238→  const [clientsLoading, setClientsLoading] = useState(false)
   239→  const [clientSearch, setClientSearch] = useState('')
   240→
   241→  const [selectedClientId, setSelectedClientId] = useState('')
   242→  const [selectedVehicleId, setSelectedVehicleId] = useState('')
   243→  const [description, setDescription] = useState('')
   244→  const [diagnosis, setDiagnosis] = useState('')
   245→  const [discount, setDiscount] = useState(0)
   246→  const [discountType, setDiscountType] = useState<'VALUE' | 'PERCENT'>('VALUE')
   247→  const [items, setItems] = useState<ServiceOrderItem[]>([
   248→    { type: 'SERVICO', description: '', quantity: 1, unitPrice: 0, totalPrice: 0, partId: null, mechanicId: null, commissionPct: 0 },
   249→  ])
   250→
   251→  // Fetch clients
   252→  useEffect(() => {
   253→    if (!open) return
   254→    const fetchClients = async () => {
   255→      setClientsLoading(true)
   256→      try {
   257→        const res = await apiFetch(`/api/clients?officeId=${officeId}&limit=100`)
   258→        if (res.success) setClients(res.data)
   259→      } catch (err) {
   260→        console.error('Error fetching clients:', err)
   261→      } finally {
   262→        setClientsLoading(false)
   263→      }
   264→    }
   265→    fetchClients()
   266→  }, [open, officeId])
   267→
   268→  // Fetch parts
   269→  useEffect(() => {
   270→    if (!open) return
   271→    const fetchParts = async () => {
   272→      try {
   273→        const res = await apiFetch(`/api/parts?officeId=${officeId}`)
   274→        if (res.success) setParts(res.data)
   275→      } catch (err) {
   276→        console.error('Error fetching parts:', err)
   277→      }
   278→    }
   279→    fetchParts()
   280→  }, [open, officeId])
   281→
   282→  // Fetch mechanics (active only)
   283→  useEffect(() => {
   284→    if (!open) return
   285→    const fetchMechanics = async () => {
   286→      try {
   287→        const res = await apiFetch(`/api/mechanics?officeId=${officeId}&active=true`)
   288→        if (res.success) setMechanics(res.data)
   289→      } catch (err) {
   290→        console.error('Error fetching mechanics:', err)
   291→      }
   292→    }
   293→    fetchMechanics()
   294→  }, [open, officeId])
   295→
   296→  // Fetch vehicles when client changes
   297→  useEffect(() => {
   298→    if (!selectedClientId) {
   299→      setVehicles([])
   300→      return
   301→    }
   302→    const fetchVehicles = async () => {
   303→      try {
   304→        const res = await apiFetch(`/api/vehicles?officeId=${officeId}&clientId=${selectedClientId}`)
   305→        if (res.success) setVehicles(res.data)
   306→      } catch (err) {
   307→        console.error('Error fetching vehicles:', err)
   308→      }
   309→    }
   310→    fetchVehicles()
   311→  }, [selectedClientId, officeId])
   312→
   313→  // Populate form when editing
   314→  useEffect(() => {
   315→    if (editingOS) {
   316→      setSelectedClientId(editingOS.clientId)
   317→      setSelectedVehicleId(editingOS.vehicleId)
   318→      setDescription(editingOS.description || '')
   319→      setDiagnosis(editingOS.diagnosis || '')
   320→      setDiscount(editingOS.totalDiscount)
   321→      setDiscountType('VALUE')
   322→      setItems(
   323→        editingOS.items.map((item) => ({
   324→          id: item.id,
   325→          type: item.type as 'SERVICO' | 'PECA',
   326→          description: item.description,
   327→          quantity: item.quantity,
   328→          unitPrice: item.unitPrice,
   329→          totalPrice: item.totalPrice,
   330→          partId: item.partId || null,
   331→          mechanicId: item.mechanicId || null,
   332→          commissionPct: item.commissionPct || 0,
   333→        }))
   334→      )
   335→    } else {
   336→      setSelectedClientId('')
   337→      setSelectedVehicleId('')
   338→      setDescription('')
   339→      setDiagnosis('')
   340→      setDiscount(0)
   341→      setDiscountType('VALUE')
   342→      setItems([
   343→        { type: 'SERVICO', description: '', quantity: 1, unitPrice: 0, totalPrice: 0, partId: null, mechanicId: null, commissionPct: 0 },
   344→      ])
   345→    }
   346→  }, [editingOS, open])
   347→
   348→  const addItem = (type: 'SERVICO' | 'PECA' = 'SERVICO') => {
   349→    if (type === 'PECA') {
   350→      setItems([...items, { type: 'PECA', description: '', quantity: 1, unitPrice: 0, totalPrice: 0, partId: null, mechanicId: null, commissionPct: 0 }])
   351→    } else {
   352→      setItems([...items, { type: 'SERVICO', description: '', quantity: 1, unitPrice: 0, totalPrice: 0, partId: null, mechanicId: null, commissionPct: 0 }])
   353→    }
   354→  }
   355→
   356→  const removeItem = (index: number) => {
   357→    if (items.length <= 1) return
   358→    setItems(items.filter((_, i) => i !== index))
   359→  }
   360→
   361→  const updateItem = (index: number, field: string, value: string | number | null) => {
   362→    const updated = [...items]
   363→    updated[index] = { ...updated[index], [field]: value }
   364→
   365→    if (field === 'quantity' || field === 'unitPrice') {
   366→      const qty = field === 'quantity' ? Number(value) : updated[index].quantity
   367→      const price = field === 'unitPrice' ? Number(value) : updated[index].unitPrice
   368→      updated[index].totalPrice = qty * price
   369→    }
   370→
   371→    // Auto-fill from part selection
   372→    if (field === 'partId' && value) {
   373→      const part = parts.find((p) => p.id === value)
   374→      if (part) {
   375→        updated[index].description = part.description
   376→        updated[index].unitPrice = part.salePrice
   377→        updated[index].totalPrice = updated[index].quantity * part.salePrice
   378→      }
   379→    }
   380→
   381→    // Auto-fill commission when mechanic is selected
   382→    if (field === 'mechanicId') {
   383→      if (value) {
   384→        const mechanic = mechanics.find((m) => m.id === value)
   385→        if (mechanic) {
   386→          updated[index].commissionPct = mechanic.commissionPct
   387→        }
   388→      } else {
   389→        updated[index].commissionPct = 0
   390→      }
   391→    }
   392→
   393→    setItems(updated)
   394→  }
   395→
   396→  const totalParts = items.filter((i) => i.type === 'PECA').reduce((sum, i) => sum + i.totalPrice, 0)
   397→  const totalServices = items.filter((i) => i.type === 'SERVICO').reduce((sum, i) => sum + i.totalPrice, 0)
   398→  const subtotal = totalParts + totalServices
   399→  const discountValue = discountType === 'PERCENT' ? (subtotal * discount) / 100 : discount
   400→  const totalAmount = Math.max(0, subtotal - discountValue)
   401→
   402→  const filteredClients = clients.filter((c) =>
   403→    !clientSearch || c.name.toLowerCase().includes(clientSearch.toLowerCase()) || (c.cpfCnpj && c.cpfCnpj.includes(clientSearch))
   404→  )
   405→
   406→  const handleSave = async () => {
   407→    if (!selectedClientId || !selectedVehicleId || items.length === 0) return
   408→
   409→    // Validate stock for PECA items
   410→    const insufficientStock = items
   411→      .filter((i) => i.type === 'PECA' && i.partId)
   412→      .find((i) => {
   413→        const part = parts.find((p) => p.id === i.partId)
   414→        return part && part.quantity < i.quantity
   415→      })
   416→    if (insufficientStock) {
   417→      const part = parts.find((p) => p.id === insufficientStock.partId)
   418→      alert(`Estoque insuficiente para "${part?.description || 'peça'}". Disponível: ${part?.quantity || 0}, Solicitado: ${insufficientStock.quantity}`)
   419→      return
   420→    }
   421→
   422→    setLoading(true)
   423→    try {
   424→      const payload = {
   425→        officeId,
   426→        clientId: selectedClientId,
   427→        vehicleId: selectedVehicleId,
   428→        description: description || undefined,
   429→        diagnosis: diagnosis || undefined,
   430→        totalDiscount: discountValue,
   431→        items: items.map((item) => ({
   432→          id: item.id,
   433→          type: item.type,
   434→          description: item.description,
   435→          quantity: item.quantity,
   436→          unitPrice: item.unitPrice,
   437→          partId: item.partId || undefined,
   438→          mechanicId: item.type === 'SERVICO' ? (item.mechanicId || undefined) : undefined,
   439→          commissionPct: item.type === 'SERVICO' && item.mechanicId ? (item.commissionPct || 0) : undefined,
   440→        })),
   441→      }
   442→
   443→      if (editingOS) {
   444→        await apiFetch(`/api/service-orders/${editingOS.id}`, {
   445→          method: 'PUT',
   446→          body: JSON.stringify(payload),
   447→        })
   448→      } else {
   449→        await apiFetch('/api/service-orders', {
   450→          method: 'POST',
   451→          body: JSON.stringify(payload),
   452→        })
   453→      }
   454→      await onSave(payload)
   455→      onClose()
   456→    } catch (err) {
   457→      console.error('Error saving OS:', err)
   458→    } finally {
   459→      setLoading(false)
   460→    }
   461→  }
   462→
   463→  return (
   464→    <Dialog open={open} onOpenChange={onClose}>
   465→      <DialogContent className="max-w-6xl max-h-[95vh] overflow-y-auto">
   466→        <DialogHeader>
   467→          <DialogTitle className="text-lg font-semibold text-zinc-900">
   468→            {editingOS ? `Editar OS ${editingOS.number}` : 'Nova Ordem de Serviço'}
   469→          </DialogTitle>
   470→        </DialogHeader>
   471→
   472→        <div className="space-y-6">
   473→          {/* Client & Vehicle */}
   474→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   475→            <div className="space-y-2">
   476→              <Label className="text-sm font-medium text-zinc-700">Cliente *</Label>
   477→              <Input
   478→                placeholder="Buscar cliente..."
   479→                value={clientSearch}
   480→                onChange={(e) => setClientSearch(e.target.value)}
   481→                className="mb-1"
   482→              />
   483→              <Select value={selectedClientId} onValueChange={(v) => { setSelectedClientId(v); setSelectedVehicleId('') }}>
   484→                <SelectTrigger className="w-full">
   485→                  <SelectValue placeholder={clientsLoading ? 'Carregando...' : 'Selecione o cliente'} />
   486→                </SelectTrigger>
   487→                <SelectContent>
   488→                  {filteredClients.map((c) => (
   489→                    <SelectItem key={c.id} value={c.id}>
   490→                      {c.name}{c.cpfCnpj ? ` - ${c.cpfCnpj}` : ''}
   491→                    </SelectItem>
   492→                  ))}
   493→                </SelectContent>
   494→              </Select>
   495→            </div>
   496→            <div className="space-y-2">
   497→              <Label className="text-sm font-medium text-zinc-700">Veículo *</Label>
   498→              <Select value={selectedVehicleId} onValueChange={setSelectedVehicleId} disabled={!selectedClientId}>
   499→                <SelectTrigger className="w-full">
   500→                  <SelectValue placeholder={selectedClientId ? 'Selecione o veículo' : 'Selecione o cliente primeiro'} />
   501→                </SelectTrigger>
   502→                <SelectContent>
   503→                  {vehicles.map((v) => (
   504→                    <SelectItem key={v.id} value={v.id}>
   505→                      {v.plate} - {v.brand || ''} {v.model || ''} {v.year || ''}
   506→                    </SelectItem>
   507→                  ))}
   508→                </SelectContent>
   509→              </Select>
   510→            </div>
   511→          </div>
   512→
   513→          {/* Description & Diagnosis */}
   514→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   515→            <div className="space-y-2">
   516→              <Label className="text-sm font-medium text-zinc-700">Descrição</Label>
   517→              <Textarea
   518→                placeholder="Descreva o serviço solicitado..."
   519→                value={description}
   520→                onChange={(e) => setDescription(e.target.value)}
   521→                rows={3}
   522→              />
   523→            </div>
   524→            <div className="space-y-2">
   525→              <Label className="text-sm font-medium text-zinc-700">Diagnóstico</Label>
   526→              <Textarea
   527→                placeholder="Diagnóstico do veículo..."
   528→                value={diagnosis}
   529→                onChange={(e) => setDiagnosis(e.target.value)}
   530→                rows={3}
   531→              />
   532→            </div>
   533→          </div>
   534→
   535→          <Separator />
   536→
   537→          {/* Items Section */}
   538→          <div className="space-y-3">
   539→            <div className="flex items-center justify-between">
   540→              <Label className="text-sm font-semibold text-zinc-700">Itens da OS</Label>
   541→              <div className="flex items-center gap-2">
   542→                <Button type="button" variant="outline" size="sm" onClick={() => addItem('SERVICO')} className="gap-1 border-blue-200 text-blue-700 hover:bg-blue-50">
   543→                  <Wrench className="w-3.5 h-3.5" /> Serviço
   544→                </Button>
   545→                <Button type="button" variant="outline" size="sm" onClick={() => addItem('PECA')} className="gap-1 border-emerald-200 text-emerald-700 hover:bg-emerald-50">
   546→                  <Package className="w-3.5 h-3.5" /> Peça
   547→                </Button>
   548→              </div>
   549→            </div>
   550→
   551→            {/* Stock deduction info banner */}
   552→            {items.some((i) => i.type === 'PECA' && i.partId) && (
   553→              <div className="flex items-center gap-2 px-3 py-2 bg-blue-50 border border-blue-200 rounded-md text-xs text-blue-700 font-medium">
   554→                <ArrowDownCircle className="w-4 h-4 shrink-0" />
   555→                Ao salvar a OS, as peças serão automaticamente deduzidas do estoque. Ao cancelar a OS, o estoque será estornado.
   556→              </div>
   557→            )}
   558→
   559→            <div className="space-y-3">
   560→              {items.map((item, index) => (
   561→                <Card key={index} className={`p-3 ${item.type === 'PECA' ? 'bg-emerald-50 border-emerald-200' : 'bg-blue-50 border-blue-200'}`}>
   562→                  <div className="flex items-center gap-2 mb-2">
   563→                    {item.type === 'PECA' ? (
   564→                      <Package className="w-4 h-4 text-emerald-600" />

   923→function OSDetail({
   924→  os,
   925→  onBack,
   926→  onStatusChange,
   927→  onEdit,
   928→}: {
   929→  os: ServiceOrder
   930→  onBack: () => void
   931→  onStatusChange: (id: string, status: OSStatus, paymentMethod?: string) => Promise<void>
   932→  onEdit: (os: ServiceOrder) => void
   933→}) {
   934→  const [statusLoading, setStatusLoading] = useState(false)
   935→  const [paymentOpen, setPaymentOpen] = useState(false)
   936→  const [showReceipt, setShowReceipt] = useState(false)
   937→  const [receiptMode, setReceiptMode] = useState<'receipt' | 'closure'>('receipt')
   938→  const { office, user } = useAuthStore()
   939→  const [receiptLogo, setReceiptLogo] = useState<string | null>(office?.logo || null)
   940→
   941→  // Client Invoice state
   942→  const [showClientInvoice, setShowClientInvoice] = useState(false)
   943→  const [clientInvoiceData, setClientInvoiceData] = useState<ClientInvoiceData | null>(null)
   944→  const [generatingNF, setGeneratingNF] = useState(false)
   945→
   946→  // WhatsApp integration state
   947→  const [waEnabled, setWaEnabled] = useState(false)
   948→  const [waSending, setWaSending] = useState(false)
   949→  const [waMessage, setWaMessage] = useState<{ type: 'success' | 'error'; text: string } | null>(null)
   950→
   951→  // Fetch WhatsApp config
   952→  useEffect(() => {
   953→    const fetchWaConfig = async () => {
   954→      if (!office) return
   955→      try {
   956→        const res = await apiFetch(`/api/whatsapp/config?officeId=${office.id}`)
   957→        if (res.success && res.data) {
   958→          setWaEnabled(res.data.enabled && res.data.configured && res.data.hasToken)
   959→        }
   960→      } catch {
   961→        // Silently handle
   962→      }
   963→    }
   964→    fetchWaConfig()
   965→  }, [office])
   966→
   967→  const handleSendWhatsApp = async (mode: 'estimate' | 'receipt') => {
   968→    if (!office) return
   969→    setWaSending(true)
   970→    setWaMessage(null)
   971→    try {
   972→      const res = await apiFetch('/api/whatsapp/send-os', {
   973→        method: 'POST',
   974→        headers: { 'Content-Type': 'application/json' },
   975→        body: JSON.stringify({
   976→          officeId: office.id,
   977→          serviceOrderId: os.id,
   978→          mode,
   979→        }),
   980→      })
   981→      if (res.success) {
   982→        const phone = res.data?.phone || os.client.phone || ''
   983→        setWaMessage({
   984→          type: 'success',
   985→          text: `${mode === 'estimate' ? 'Orçamento' : 'Recibo'} enviado via WhatsApp para ${phone}`,
   986→        })
   987→      } else {
   988→        setWaMessage({
   989→          type: 'error',
   990→          text: res.error || 'Erro ao enviar via WhatsApp',
   991→        })
   992→      }
   993→    } catch {
   994→      setWaMessage({ type: 'error', text: 'Erro de conexão ao enviar via WhatsApp' })
   995→    } finally {
   996→      setWaSending(false)
   997→      setTimeout(() => setWaMessage(null), 5000)
   998→    }
   999→  }
  1000→
  1001→  // Resolve logo URL for API route serving
  1002→  const resolveLogoUrl = (logo: string | null | undefined): string | null => {
  1003→    if (!logo) return null
  1004→    if (logo.startsWith('http') || logo.startsWith('data:')) return logo
  1005→    return logo.replace(/^\/uploads\//, '/api/uploads/')
  1006→  }
  1007→
  1008→  // Fetch fresh logo when receipt opens
  1009→  const handleOpenReceipt = async (mode: 'receipt' | 'closure') => {
  1010→    setReceiptMode(mode)
  1011→    try {
  1012→      const data = await apiFetch(`/api/auth/me`)
  1013→      if (data.success && data.data?.office?.logo) {
  1014→        setReceiptLogo(resolveLogoUrl(data.data.office.logo))
  1015→      } else {
  1016→        setReceiptLogo(resolveLogoUrl(office?.logo))
  1017→      }
  1018→    } catch {
  1019→      setReceiptLogo(resolveLogoUrl(office?.logo))
  1020→    }
  1021→    setShowReceipt(true)
  1022→  }
  1023→
  1024→  const handleStatusChange = async (newStatus: OSStatus, paymentMethod?: string) => {
  1025→    setStatusLoading(true)
  1026→    try {
  1027→      await onStatusChange(os.id, newStatus, paymentMethod)
  1028→    } finally {
  1029→      setStatusLoading(false)
  1030→      setPaymentOpen(false)
  1031→    }
  1032→  }
  1033→
  1034→  // Emit NF for this OS
  1035→  const handleEmitNF = async () => {
  1036→    if (!office) return
  1037→    setGeneratingNF(true)
  1038→    try {
  1039→      const res = await apiFetch('/api/client-invoices', {
  1040→        method: 'POST',
  1041→        body: JSON.stringify({
  1042→          officeId: office.id,
  1043→          type: 'SERVICO',
  1044→          serviceOrderId: os.id,
  1045→        }),
  1046→      })
  1047→      if (res.success && res.data) {
  1048→        const detailRes = await apiFetch(`/api/client-invoices/${res.data.id}?officeId=${office.id}`)
  1049→        if (detailRes.success && detailRes.data) {
  1050→          setClientInvoiceData(detailRes.data as ClientInvoiceData)
  1051→          setShowClientInvoice(true)
  1052→        }
  1053→      } else {
  1054→        alert(res.error || 'Erro ao emitir nota fiscal')
  1055→      }
  1056→    } catch {
  1057→      alert('Erro de conexão ao emitir nota fiscal')
  1058→    } finally {
  1059→      setGeneratingNF(false)
  1060→    }
  1061→  }
  1062→
  1063→  const pecaItems = os.items.filter((i) => i.type === 'PECA')
  1064→  const servicoItems = os.items.filter((i) => i.type === 'SERVICO')
  1065→
  1066→  return (
  1067→    <div className="space-y-6">
  1068→      {/* Header */}
  1069→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">

  1071→          <Button variant="ghost" size="sm" onClick={onBack} className="gap-1">
  1072→            <ArrowLeft className="w-4 h-4" /> Voltar
  1073→          </Button>
  1074→          <div>
  1075→            <h2 className="text-xl font-bold text-zinc-900">{os.number}</h2>
  1076→            <p className="text-sm text-zinc-500">
  1077→              Criada em {formatDate(os.createdAt)}
  1078→            </p>
  1079→          </div>
  1080→        </div>
  1081→        <div className="flex items-center gap-2">
  1082→          <StatusBadge status={os.status} />
  1083→          {os.status !== 'PAGA' && os.status !== 'CANCELADA' && (
  1084→            <Button variant="outline" size="sm" onClick={() => onEdit(os)}>
  1085→              Editar
  1086→            </Button>
  1087→          )}
  1088→        </div>
  1089→      </div>
  1090→
  1091→      {/* Status Timeline */}
  1092→      <Card className="p-6">
  1093→        <StatusTimeline status={os.status} />
