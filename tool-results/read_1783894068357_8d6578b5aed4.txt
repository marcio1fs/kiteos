    25→    25→/**
    26→    26→ * Extrai valor de tag XML usando regex (mesmo padrão do xml-parser)
    27→    27→ */
    28→    28→function extractValue(xml: string, tag: string): string | null {
    29→    29→  const cleanTag = tag.replace(/[\w]+:/g, '')
    30→    30→  const patterns = [
    31→    31→    new RegExp(`<[\\w]*:?${escapeRegex(cleanTag)}[^>]*>([^<]+)</[\\w]*:?${escapeRegex(cleanTag)}>`, 'i'),
    32→    32→    new RegExp(`<${escapeRegex(cleanTag)}[^>]*>([^<]+)</${escapeRegex(cleanTag)}>`, 'i'),
    33→    33→  ]
    34→    34→  for (const pattern of patterns) {
    35→    35→    const match = xml.match(pattern)
    36→    36→    if (match && match[1]) return match[1].trim()
    37→    37→  }
    38→    38→  return null
    39→    39→}
    40→    40→
    41→    41→function escapeRegex(str: string): string {
    42→    42→  return str.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
    43→    43→}
    44→    44→
    45→    45→/**
    46→    46→ * Extrai bloco XML
    47→    47→ */
    48→    48→function extractBlock(xml: string, tag: string): string | null {
    49→    49→  const cleanTag = tag.replace(/[\w]+:/g, '')
    50→    50→  const pattern = new RegExp(
    51→    51→    `<[\\w]*:?${escapeRegex(cleanTag)}[^>]*>([\\s\\S]*?)</[\\w]*:?${escapeRegex(cleanTag)}>`,
    52→    52→    'i'
    53→    53→  )
    54→    54→  const match = xml.match(pattern)
    55→    55→  return match ? match[1] : null
    56→    56→}

    61→    61→function mapCStatToStatus(cStat: string): SefazConsultaResult['nfeStatus'] {
    62→    62→  const authorized = ['100', '150'] // Autorizado o uso da NF-e / Autorizado o uso da NF-e
    63→    63→  const cancelled = ['101', '151', '218'] // Cancelamento / Cancelamento por substituição
    64→    64→  const denied = ['110', '205', '301', '302'] // Uso Denegado
    65→    65→  const unused = ['102', '217', '562'] // Inutilização
    66→    66→
    67→    67→  if (authorized.includes(cStat)) return 'AUTORIZADA'
    68→    68→  if (cancelled.includes(cStat)) return 'CANCELADA'
    69→    69→  if (denied.includes(cStat)) return 'DENEGADA'
    70→    70→  if (unused.includes(cStat)) return 'INUTILIZADA'
    71→    71→  if (['516', '517', '518', '563', '564', '565', '567', '568', '578', '579', '580'].includes(cStat)) return 'NAO_ENCONTRADA'
    72→    72→
    73→    73→  return 'ERRO'
    74→    74→}

    76→    76→/**
    77→    77→ * Parse da resposta de Consulta Protocolo
    78→    78→ * Extrai dados do envelope SOAP retornado pela SEFAZ
    79→    79→ */
    80→    80→export function parseConsultaProtocoloResponse(soapResponse: string): SefazConsultaResult {
    81→    81→  // Extrair corpo do SOAP
    82→    82→  const body = extractBlock(soapResponse, 'soap:Body') || extractBlock(soapResponse, 'Body') || soapResponse
    83→    83→
    84→    84→  // Extrair retConsSitNFe
    85→    85→  const retConsSitNFe = extractBlock(body, 'retConsSitNFe') || body
    86→    86→
    87→    87→  const cStat = extractValue(retConsSitNFe, 'cStat') || ''
    88→    88→  const xMotivo = extractValue(retConsSitNFe, 'xMotivo') || 'Resposta não identificada'
    89→    89→  const chNFe = extractValue(retConsSitNFe, 'chNFe') || ''
    90→    90→  const cUF = extractValue(retConsSitNFe, 'cUF')
    91→    91→
    92→    92→  // Extrair protocolo de autorização
    93→    93→  let protocolo: string | null = null
    94→    94→  let dhRecbto: string | null = null
    95→    95→  let fornecedorCnpj: string | null = null
    96→    96→
    97→    97→  // Buscar protocolo dentro de protNFe
    98→    98→  const protNFeBlock = extractBlock(retConsSitNFe, 'protNFe')
    99→    99→  if (protNFeBlock) {
   100→   100→    protocolo = extractValue(protNFeBlock, 'nProt')
   101→   101→    dhRecbto = extractValue(protNFeBlock, 'dhRecbto')
   102→   102→
   103→   103→    // Tentar extrair CNPJ do emitente do XML do protocolo
   104→   104→    const infProtBlock = extractBlock(protNFeBlock, 'infProt')
   105→   105→    if (infProtBlock) {
   106→   106→      // Se houver XML dentro do protocolo, extrair emit/CNPJ
   107→   107→      const emitCnpj = extractValue(infProtBlock, 'CNPJ')
   108→   108→      if (emitCnpj) fornecedorCnpj = emitCnpj
   109→   109→    }
   110→   110→  }
   111→   111→
   112→   112→  return {
   113→   113→    cStat,
   114→   114→    xMotivo,
   115→   115→    protocolo,
   116→   116→    dhRecbto,
   117→   117→    chNFe,
   118→   118→    cUF,
   119→   119→    nfeStatus: mapCStatToStatus(cStat),
   120→   120→    xmlContent: null, // Consulta protocolo não retorna XML completo
   121→   121→    fornecedorCnpj,
   122→   122→  }
   123→   123→}

   125→   125→/**
   126→   126→ * Parse da resposta de Download NF-e
   127→   127→ * Extrai XML autorizado da resposta
   128→   128→ */
   129→   129→export function parseDownloadNfeResponse(soapResponse: string): SefazDownloadResult {
   130→   130→  const body = extractBlock(soapResponse, 'soap:Body') || extractBlock(soapResponse, 'Body') || soapResponse
   131→   131→
   132→   132→  const retDownloadNFe = extractBlock(body, 'retDownloadNFe') || body
   133→   133→
   134→   134→  const cStat = extractValue(retDownloadNFe, 'cStat') || ''
   135→   135→  const xMotivo = extractValue(retDownloadNFe, 'xMotivo') || 'Resposta não identificada'
   136→   136→  const chNFe = extractValue(retDownloadNFe, 'chNFe') || ''
   137→   137→
   138→   138→  // Extrair XML da NF-e
   139→   139→  let xmlContent: string | null = null
   140→   140→  const procNFeBlock = extractBlock(retDownloadNFe, 'procNFe')
   141→   141→  if (procNFeBlock) {
   142→   142→    // O procNFe contém o NFe + protNFe completo
   143→   143→    // Precisamos reconstruir o XML com a tag raiz
   144→   144→    xmlContent = `<?xml version="1.0" encoding="UTF-8"?><nfeProc xmlns="http://www.portalfiscal.inf.br/nfe">${procNFeBlock}</nfeProc>`
   145→   145→  }
   146→   146→
   147→   147→  // Se não encontrou procNFe, tentar extrair NFe diretamente
   148→   148→  if (!xmlContent) {
   149→   149→    const nfeBlock = extractBlock(retDownloadNFe, 'NFe')
   150→   150→    if (nfeBlock) {
   151→   151→      xmlContent = `<?xml version="1.0" encoding="UTF-8"?><NFe xmlns="http://www.portalfiscal.inf.br/nfe">${nfeBlock}</NFe>`
   152→   152→    }
   153→   153→  }
   154→   154→
   155→   155→  return {
   156→   156→    cStat,
   157→   157→    xMotivo,
   158→   158→    chNFe,
   159→   159→    xmlContent,
   160→   160→  }
   161→   161→}
