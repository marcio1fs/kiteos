    60→/**
    61→ * Format a number using Brazilian decimal separator (comma)
    62→ * Removes trailing zeros after decimal
    63→ */
    64→function formatNumber(value: number | null | undefined, decimals: number = 2): string {
    65→  if (value === null || value === undefined || isNaN(value)) return ''
    66→  const fixed = value.toFixed(decimals)
    67→  // Replace dot with comma for Brazilian format
    68→  return fixed.replace('.', ',')
    69→}
    70→
    71→/**
    72→ * Format a Date to DDMMYYYY string
    73→ */
    74→function formatDate(date: Date | null | undefined): string {
    75→  if (!date) return ''
    76→  const d = new Date(date)
    77→  const day = String(d.getDate()).padStart(2, '0')
    78→  const month = String(d.getMonth() + 1).padStart(2, '0')
    79→  const year = String(d.getFullYear())
    80→  return `${day}${month}${year}`
    81→}
    82→
    83→/**
    84→ * Remove non-digit characters from CNPJ/CPF
    85→ */
    86→function onlyDigits(value: string | null | undefined): string {
    87→  if (!value) return ''
    88→  return value.replace(/\D/g, '')
    89→}
    90→
    91→/**
    92→ * Build a SPED line from fields: |field1|field2|...|
    93→ */
    94→function buildLine(fields: (string | number | null | undefined)[]): string {
    95→  const parts = fields.map(f => {
    96→    if (f === null || f === undefined) return ''
    97→    return String(f)
    98→  })
    99→  return '|' + parts.join('|') + '|'
   100→}

   102→/**
   103→ * Parse competencia AAAAMM to start/end dates
   104→ */
   105→function parseCompetencia(competencia: string): { dataInicio: Date; dataFim: Date } {
   106→  const ano = parseInt(competencia.substring(0, 4))
   107→  const mes = parseInt(competencia.substring(4, 6))
   108→
   109→  const dataInicio = new Date(ano, mes - 1, 1, 0, 0, 0, 0)
   110→  const dataFim = new Date(ano, mes, 0, 23, 59, 59, 999) // last day of month
   111→
   112→  return { dataInicio, dataFim }
   113→}

   115→/**
   116→ * Determine the COD_VER (código da versão) from versao string
   117→ */
   118→function getCodVer(versao: string): string {
   119→  const versionMap: Record<string, string> = {
   120→    '019': '019', // Layout 19 (vigente a partir de 2024)
   121→    '018': '018', // Layout 18 (vigente a partir de 2023)
   122→    '017': '017', // Layout 17 (vigente a partir de 2022)
   123→    '016': '016',
   124→    '015': '015',
   125→    '014': '014',
   126→    '013': '013',
   127→    '012': '012',
   128→    '011': '011',
   129→    '010': '010',
   130→    '009': '009',
   131→    '008': '008',
   132→    '007': '007',
   133→    '006': '006',
   134→    '005': '005',
   135→  }
   136→  return versionMap[versao] || '019'
   137→}
   138→
   139→/**
   140→ * Determine COD_FIN (finalidade do arquivo):
   141→ * 0 - Remessa do arquivo original
   142→ * 1 - Remessa do arquivo substituto
   143→ */
   144→function getCodFin(): string {
   145→  return '0' // Original submission
   146→}
   147→
   148→/**
   149→ * Get IND_PERFIL (perfil de apresentação):
   150→ * A - Perfil A
   151→ * B - Perfil B
   152→ * C - Perfil C
   153→ * Empty - Não enquadrado
   154→ */
   155→function getIndPerfil(): string {
   156→  return '' // For Simples Nacional workshops, typically not assigned
   157→}
   158→
   159→/**
   160→ * Get IND_ATIV (indicador de atividade):
   161→ * 0 - Industrial ou equiparado a industrial
   162→ * 1 - Outros
   163→ */
   164→function getIndAtiv(): string {
   165→  return '1' // Mechanic workshops are typically "Outros"
   166→}

   172→/**
   173→ * BLOCO 0 - Abertura, Identificação e Referências
   174→ */
   175→async function generateBloco0(
   176→  config: SpedGeneratorConfig,
   177→  dataInicio: Date,
   178→  dataFim: Date
   179→): Promise<SpedLine[]> {
   180→  const lines: SpedLine[] = []
   181→  let ordem = 0
   182→
   183→  // Get office data
   184→  const office = await db.office.findUnique({
   185→    where: { id: config.officeId },
   186→  })
   187→  if (!office) {
   188→    throw new Error('Escritório não encontrado')
   189→  }
   190→
   191→  // |0000| - Abertura do arquivo e identificação da entidade
   192→  const codVer = getCodVer(config.versao || '019')
   193→  const codFin = getCodFin()
   194→  const dtIni = formatDate(dataInicio)
   195→  const dtFin = formatDate(dataFim)
   196→  const cnpj = onlyDigits(office.cnpj)
   197→  const nome = office.name
   198→  const uf = office.state || ''
   199→  const ie = '' // Will need to be stored in office model; using empty
   200→  const codMun = '' // IBGE code for municipality
   201→  const im = '' // Inscrição Municipal
   202→  const indPerfil = getIndPerfil()
   203→  const indAtiv = getIndAtiv()
   204→
   205→  lines.push({
   206→    registro: '0000',
   207→    bloco: '0',
   208→    linha: buildLine(['0000', codVer, codFin, dtIni, dtFin, cnpj, nome, cnpj.substring(0, 8), uf, ie, codMun, im, indPerfil, indAtiv]),
   209→    ordem: ordem++,
   210→  })
   211→
   212→  // |0001| - Abertura do Bloco 0
   213→  // IND_MOV: 0 = com dados, 1 = sem dados
   214→  lines.push({
   215→    registro: '0001',
   216→    bloco: '0',
   217→    linha: buildLine(['0001', '0']),
   218→    ordem: ordem++,
   219→  })
   220→
   221→  // |0005| - Dados complementares da entidade
   222→  const fantasia = office.name
   223→  const cep = onlyDigits(office.zipCode)
   224→  const end = office.address || ''
   225→  const num = ''
   226→  const compl = ''
   227→  const bairro = ''
   228→  const fone = onlyDigits(office.phone)
   229→  const fax = ''
   230→  const email = office.email
   231→
   232→  lines.push({
   233→    registro: '0005',
   234→    bloco: '0',
   235→    linha: buildLine(['0005', fantasia, cep, end, num, compl, bairro, fone, fax, email]),
   236→    ordem: ordem++,
   237→  })
   238→
   239→  // |0100| - Identificação do estabelecimento
   240→  // Contador / responsável pela escrita fiscal
   241→  lines.push({
   242→    registro: '0100',
   243→    bloco: '0',
   244→    linha: buildLine(['0100', 'CONTADOR', '', '', '', '', '', '9999']),
   245→    ordem: ordem++,
   246→  })
   247→
   248→  // |0150| - Cadastro do participante (fornecedores e clientes)
   249→  const fornecedores = await db.supplier.findMany({
   250→    where: {
   251→      officeId: config.officeId,
   252→      deleted: false,
   253→    },
   254→  })
   255→
   256→  const clientes = await db.client.findMany({
   257→    where: {
   258→      officeId: config.officeId,
   259→      deleted: false,
   260→    },
   261→  })
   262→
   263→  const participantes = new Map<string, { codigo: string; nome: string; cnpj: string; ie: string; uf: string; cep: string; endereco: string; numero: string; complemento: string; bairro: string; fone: string; pais: string }>()
   264→
   265→  // Add suppliers
   266→  for (const forn of fornecedores) {
   267→    const cnpjLimpo = onlyDigits(forn.cnpj)
   268→    if (!participantes.has(cnpjLimpo)) {
   269→      participantes.set(cnpjLimpo, {
   270→        codigo: `FOR${cnpjLimpo.substring(0, 8)}`,
   271→        nome: forn.razaoSocial,
   272→        cnpj: cnpjLimpo,
   273→        ie: forn.ie || '',
   274→        uf: forn.state || '',
   275→        cep: onlyDigits(forn.zipCode),
   276→        endereco: forn.address || '',
   277→        numero: '',
   278→        complemento: '',
   279→        bairro: '',
   280→        fone: onlyDigits(forn.phone),
   281→        pais: '105', // Brasil
   282→      })
   283→    }
   284→  }
   285→
   286→  // Add clients
   287→  for (const cli of clientes) {
   288→    const cpfCnpjLimpo = onlyDigits(cli.cpfCnpj)
   289→    if (cpfCnpjLimpo && !participantes.has(cpfCnpjLimpo)) {
   290→      participantes.set(cpfCnpjLimpo, {
   291→        codigo: `CLI${cpfCnpjLimpo.substring(0, 8)}`,
   292→        nome: cli.name,
   293→        cnpj: cpfCnpjLimpo,
   294→        ie: '',
   295→        uf: cli.state || '',
   296→        cep: onlyDigits(cli.zipCode),
   297→        endereco: cli.address || '',
   298→        numero: '',
   299→        complemento: '',
   300→        bairro: '',
   301→        fone: onlyDigits(cli.phone),
   302→        pais: '105', // Brasil
   303→      })
   304→    }
   305→  }
   306→
   307→  for (const [, part] of participantes) {
   308→    const isCnpj = part.cnpj.length === 14
   309→    lines.push({
   310→      registro: '0150',
   311→      bloco: '0',
   312→      linha: buildLine([
   313→        '0150',
   314→        part.codigo,
   315→        part.nome,
   316→        isCnpj ? '00' : '01', // COD_PAIS: 00=CNPJ, 01=CPF
   317→        part.cnpj,
   318→        part.ie,
   319→        part.uf,
   320→        part.cep,
   321→        part.endereco,
   322→        part.numero,
   323→        part.complemento,
   324→        part.bairro,
   325→        part.fone,
   326→        part.pais,
   327→      ]),
   328→      ordem: ordem++,
   329→    })
   330→  }
   331→
   332→  // |0190| - Identificação das unidades de medida
   333→  const unidades = new Set<string>()
   334→  // Get units from parts
   335→  const pecas = await db.part.findMany({
   336→    where: {
   337→      officeId: config.officeId,
   338→      deleted: false,
   339→    },
   340→    select: { unit: true },
   341→  })
   342→  for (const p of pecas) {
   343→    if (p.unit) unidades.add(p.unit)
   344→  }
   345→  // Add default
   346→  unidades.add('UN')
   347→
   348→  for (const unidade of unidades) {
   349→    lines.push({
   350→      registro: '0190',
   351→      bloco: '0',
   352→      linha: buildLine(['0190', unidade, unidade === 'UN' ? 'UNIDADE' : unidade]),
   353→      ordem: ordem++,
   354→    })
   355→  }
   356→
   357→  // |0200| - Identificação dos itens (produtos)
   358→  const itens = await db.part.findMany({
   359→    where: {
   360→      officeId: config.officeId,
   361→      deleted: false,
   362→    },
   363→    orderBy: { sku: 'asc' },
   364→  })
   365→
   366→  for (const item of itens) {
   367→    lines.push({
   368→      registro: '0200',
   369→      bloco: '0',
   370→      linha: buildLine([
   371→        '0200',
   372→        item.sku,                     // COD_ITEM
   373→        item.description,             // DESCR_ITEM
   374→        '',                           // COD_BARRA
   375→        item.ncm || '',               // COD_NCM
   376→        '',                           // EX_IPI
   377→        item.cfop || '',              // COD_GEN (gênero)
   378→        '',                           // COD_LST
   379→        item.cst || '',               // ALIQ_ICMS
   380→        item.csosn || '',             // CEST (actually CSOSN, but field position)
   381→        item.unit,                    // UNID_INV
   382→        formatNumber(item.quantity, 3), // QTD_INV
   383→        formatNumber(item.costPrice),   // VAL_INV
   384→      ]),
   385→      ordem: ordem++,
   386→    })
   387→  }
   388→
   389→  // |0990| - Encerramento do Bloco 0
   390→  // Count will be updated later
   391→  lines.push({
   392→    registro: '0990',
   393→    bloco: '0',
   394→    linha: buildLine(['0990', '0']), // placeholder count
   395→    ordem: ordem++,
   396→  })
   397→
   398→  // Update Block 0 count
   399→  const block0Count = lines.filter(l => l.bloco === '0').length
   400→  const linha0990 = lines.find(l => l.registro === '0990')
   401→  if (linha0990) {
   402→    linha0990.linha = buildLine(['0990', String(block0Count)])
   403→  }
   404→
   405→  return lines
   406→}

   408→/**
   409→ * BLOCO C - Documentos Fiscais I (Mercadorias)
   410→ */
   411→async function generateBlocoC(
   412→  config: SpedGeneratorConfig,
   413→  dataInicio: Date,
   414→  dataFim: Date
   415→): Promise<{ lines: SpedLine[]; cfopResumos: CfopResumo[] }> {
   416→  const lines: SpedLine[] = []
   417→  const cfopResumos: CfopResumo[] = []
   418→  let ordem = 0
   419→
   420→  // |C001| - Abertura do Bloco C
   421→  // Check if there are invoices in the period
   422→  const invoiceCount = await db.invoice.count({
   423→    where: {
   424→      officeId: config.officeId,
   425→      deleted: false,
   426→      dataEmissao: { gte: dataInicio, lte: dataFim },
   427→    },
   428→  })
   429→
   430→  lines.push({
   431→    registro: 'C001',
   432→    bloco: 'C',
   433→    linha: buildLine(['C001', invoiceCount > 0 ? '0' : '1']),
   434→    ordem: ordem++,
   435→  })
   436→
   437→  if (invoiceCount > 0) {
   438→    // Get invoices with items for the period
   439→    const invoices = await db.invoice.findMany({
   440→      where: {
   441→        officeId: config.officeId,
   442→        deleted: false,
   443→        dataEmissao: { gte: dataInicio, lte: dataFim },
   444→      },
   445→      include: {
   446→        items: {
   447→          where: { deleted: false },
   448→          orderBy: { createdAt: 'asc' },
   449→        },
   450→        supplier: true,
   451→      },
   452→      orderBy: { dataEmissao: 'asc' },
   453→    })
   454→
   455→    // Also get client invoices (saída)
   456→    const clientInvoices = await db.clientInvoice.findMany({
   457→      where: {
   458→        officeId: config.officeId,
   459→        deleted: false,
   460→        dataEmissao: { gte: dataInicio, lte: dataFim },
   461→        status: 'EMITIDA',
   462→      },
   463→      include: {
   464→        items: {
   465→          where: { deleted: false },
   466→          orderBy: { createdAt: 'asc' },
   467→        },
   468→      },
   469→      orderBy: { dataEmissao: 'asc' },
   470→    })
   471→
   472→    // Process NF-e de entrada (from suppliers)
   473→    for (const inv of invoices) {
   474→      const indOper = '0' // 0=Entrada
   475→      const indEmit = '1' // 1=Terceiros
   476→      const codPart = `FOR${onlyDigits(inv.supplier.cnpj).substring(0, 8)}`
   477→      const codMod = '55' // NF-e
   478→      const codSit = '00' // Documento regular
   479→      const serie = inv.serie
   480→      const numDoc = inv.numero
   481→      const dtDoc = formatDate(inv.dataEmissao)
   482→      const dtES = inv.dataEntrada ? formatDate(inv.dataEntrada) : formatDate(inv.dataEmissao)
   483→      const vlDoc = formatNumber(inv.totalNf)
   484→      const indPgto = '' // Indicador de pagamento
   485→      const vlDesc = formatNumber(inv.desconto)
   486→      const vlAbatNT = '' // Abatimento não tributado
   487→      const vlMerc = formatNumber(inv.subtotal)
   488→      const indFrt = '' // Indicador de frete
   489→      const vlFrt = formatNumber(inv.frete)
   490→      const vlSeg = formatNumber(inv.seguro)
   491→      const vlOutDa = '' // Outras despesas acessórias
   492→      const vlBCICMS = formatNumber(inv.baseIcms)
   493→      const vlICMS = formatNumber(inv.valorIcms)
   494→      const vlBCICMSST = ''
   495→      const vlICMSST = ''
   496→      const vlIPI = formatNumber(inv.valorIpi)
   497→      const vlPIS = formatNumber(inv.valorPis)
   498→      const vlCOFINS = formatNumber(inv.valorCofins)
   499→      const vlPISST = ''
   500→      const vlCOFINSST = ''
   501→
   502→      lines.push({
   503→        registro: 'C100',
   504→        bloco: 'C',
   505→        linha: buildLine([
   506→          'C100', indOper, indEmit, codPart, codMod, codSit,
   507→          serie, numDoc, dtDoc, dtES, vlDoc, indPgto,
   508→          vlDesc, vlAbatNT, vlMerc, indFrt, vlFrt, vlSeg,
   509→          vlOutDa, vlBCICMS, vlICMS, vlBCICMSST, vlICMSST,
   510→          vlIPI, vlPIS, vlCOFINS, vlPISST, vlCOFINSST,
   511→        ]),
   512→        ordem: ordem++,
   513→      })
   514→
   515→      // |C170| - Itens do documento
   516→      let itemNum = 1
   517→      for (const item of inv.items) {
   518→        const numItem = String(itemNum)
   519→        const codItem = item.sku || item.partId || `ITEM${itemNum}`
   520→        const descrCompl = item.descricao
   521→        const qtd = formatNumber(item.quantidade, 3)
   522→        const und = item.unidade
   523→        const vlItem = formatNumber(item.valorTotal)
   524→        const vlDesc = ''
   525→        const indMov = '' // Movimentação física
   526→        const cst = item.cst || item.csosn || ''
   527→        const cfop = item.cfop
   528→        const codNat = '' // Código da natureza da operação
   529→        const vlBCICMS = formatNumber(item.baseIcms)
   530→        const aliqICMS = item.baseIcms > 0 && item.valorIcms > 0 ? formatNumber((item.valorIcms / item.baseIcms) * 100) : ''
   531→        const vlICMS = formatNumber(item.valorIcms)
   532→        const vlBCICMSST = ''
   533→        const aliqST = ''
   534→        const vlICMSST = ''
   535→        const indApur = '' // Indicador de apuração do IPI
   536→        const cstIPI = ''
   537→        const codEnq = ''
   538→        const vlBCIPI = ''
   539→        const aliqIPI = item.valorIpi > 0 && item.valorTotal > 0 ? formatNumber((item.valorIpi / item.valorTotal) * 100) : ''
   540→        const vlIPI = formatNumber(item.valorIpi)
   541→        const cstPIS = ''
   542→        const vlBCPIS = ''
   543→        const aliqPIS = ''
   544→        const quantBCPIS = ''
   545→        const vlPIS = formatNumber(item.valorPis)
   546→        const cstCOFINS = ''
   547→        const vlBCCOFINS = ''
   548→        const aliqCOFINS = ''
   549→        const quantBCCOFINS = ''
   550→        const vlCOFINS = formatNumber(item.valorCofins)
   551→        const codCTA = ''
   552→
   553→        lines.push({
   554→          registro: 'C170',
   555→          bloco: 'C',
   556→          linha: buildLine([
   557→            'C170', numItem, codItem, descrCompl, qtd, und,
   558→            vlItem, vlDesc, indMov, cst, cfop, codNat,
   559→            vlBCICMS, aliqICMS, vlICMS, vlBCICMSST, aliqST,
   560→            vlICMSST, indApur, cstIPI, codEnq, vlBCIPI, aliqIPI,
   561→            vlIPI, cstPIS, vlBCPIS, aliqPIS, quantBCPIS, vlPIS,
   562→            cstCOFINS, vlBCCOFINS, aliqCOFINS, quantBCCOFINS, vlCOFINS,
   563→            codCTA,
   564→          ]),
   565→          ordem: ordem++,
   566→        })
   567→
   568→        // Aggregate CFOP resumo
   569→        if (cfop) {
   570→          const existing = cfopResumos.find(r => r.cfop === cfop && r.cst === (cst || ''))
   571→          if (existing) {
   572→            existing.baseIcms += item.baseIcms
   573→            existing.valorIcms += item.valorIcms
   574→            existing.valorIpi += item.valorIpi
   575→            existing.valorPis += item.valorPis
   576→            existing.valorCofins += item.valorCofins
   577→            existing.valorTotal += item.valorTotal
   578→          } else {
   579→            cfopResumos.push({
   580→              cfop,
   581→              cst: cst || '',
   582→              baseIcms: item.baseIcms,
   583→              valorIcms: item.valorIcms,
   584→              valorIpi: item.valorIpi,
   585→              valorPis: item.valorPis,
   586→              valorCofins: item.valorCofins,
   587→              valorTotal: item.valorTotal,
   588→            })
   589→          }
   590→        }
   591→
   592→        itemNum++
   593→      }
   594→    }
   595→
   596→    // Process NF-e de saída (client invoices)
   597→    for (const cinv of clientInvoices) {
   598→      const indOper = '1' // 1=Saída
   599→      const indEmit = '0' // 0=Emissão própria
   600→      const codPart = cinv.destCpfCnpj ? `CLI${onlyDigits(cinv.destCpfCnpj).substring(0, 8)}` : ''

   602→      const codSit = '00' // Documento regular

   604→      const numDoc = cinv.number
   605→      const dtDoc = formatDate(cinv.dataEmissao)
   606→      const dtES = formatDate(cinv.dataEmissao)
   607→      const vlDoc = formatNumber(cinv.totalAmount)
   608→      const indPgto = ''
   609→      const vlDesc = formatNumber(cinv.desconto)
   610→      const vlAbatNT = ''
   611→      const vlMerc = formatNumber(cinv.subtotal)
   612→      const indFrt = ''
   613→      const vlFrt = ''
   614→      const vlSeg = ''
   615→      const vlOutDa = ''
   616→      const vlBCICMS = formatNumber(cinv.baseIcms)
   617→      const vlICMS = formatNumber(cinv.valorIcms)
   618→      const vlBCICMSST = ''
   619→      const vlICMSST = ''
   620→      const vlIPI = formatNumber(cinv.valorIpi)
   621→      const vlPIS = formatNumber(cinv.valorPis)
   622→      const vlCOFINS = formatNumber(cinv.valorCofins)
   623→      const vlPISST = ''
   624→      const vlCOFINSST = ''
   625→
   626→      lines.push({
   627→        registro: 'C100',
   628→        bloco: 'C',
   629→        linha: buildLine([
   630→          'C100', indOper, indEmit, codPart, codMod, codSit,
   631→          serie, numDoc, dtDoc, dtES, vlDoc, indPgto,
   632→          vlDesc, vlAbatNT, vlMerc, indFrt, vlFrt, vlSeg,
   633→          vlOutDa, vlBCICMS, vlICMS, vlBCICMSST, vlICMSST,
   634→          vlIPI, vlPIS, vlCOFINS, vlPISST, vlCOFINSST,
   635→        ]),
   636→        ordem: ordem++,
   637→      })
   638→
   639→      // |C170| - Itens do documento
   640→      let itemNum = 1
   641→      for (const item of cinv.items) {
   642→        const numItem = String(itemNum)
   643→        const codItem = item.cfop || `ITEM${itemNum}`
   644→        const descrCompl = item.descricao
   645→        const qtd = formatNumber(item.quantidade, 3)
   646→        const und = item.unidade
   647→        const vlItem = formatNumber(item.valorTotal)
   648→        const vlDesc = ''
   649→        const indMov = ''
   650→        const cst = item.cst || item.csosn || ''
   651→        const cfop = item.cfop || ''
   652→        const codNat = ''
   653→        const vlBCICMS = formatNumber(item.baseIcms)
   654→        const aliqICMS = item.baseIcms > 0 && item.valorIcms > 0 ? formatNumber((item.valorIcms / item.baseIcms) * 100) : ''
   655→        const vlICMS = formatNumber(item.valorIcms)
   656→        const vlBCICMSST = ''
   657→        const aliqST = ''
   658→        const vlICMSST = ''
   659→        const indApur = ''
   660→        const cstIPI = ''
   661→        const codEnq = ''
   662→        const vlBCIPI = ''
   663→        const aliqIPI = item.valorIpi > 0 && item.valorTotal > 0 ? formatNumber((item.valorIpi / item.valorTotal) * 100) : ''
   664→        const vlIPI = formatNumber(item.valorIpi)
   665→        const cstPIS = ''
   666→        const vlBCPIS = ''
   667→        const aliqPIS = ''
   668→        const quantBCPIS = ''
   669→        const vlPIS = formatNumber(item.valorPis)
   670→        const cstCOFINS = ''
   671→        const vlBCCOFINS = ''
   672→        const aliqCOFINS = ''
   673→        const quantBCCOFINS = ''
   674→        const vlCOFINS = formatNumber(item.valorCofins)
   675→        const codCTA = ''
   676→
   677→        lines.push({
   678→          registro: 'C170',
   679→          bloco: 'C',
   680→          linha: buildLine([
   681→            'C170', numItem, codItem, descrCompl, qtd, und,
   682→            vlItem, vlDesc, indMov, cst, cfop, codNat,
   683→            vlBCICMS, aliqICMS, vlICMS, vlBCICMSST, aliqST,
   684→            vlICMSST, indApur, cstIPI, codEnq, vlBCIPI, aliqIPI,
   685→            vlIPI, cstPIS, vlBCPIS, aliqPIS, quantBCPIS, vlPIS,
   686→            cstCOFINS, vlBCCOFINS, aliqCOFINS, quantBCCOFINS, vlCOFINS,
   687→            codCTA,
   688→          ]),
   689→          ordem: ordem++,
   690→        })
   691→
   692→        // Aggregate CFOP resumo for saída
   693→        if (cfop) {
   694→          const existing = cfopResumos.find(r => r.cfop === cfop && r.cst === (cst || ''))
   695→          if (existing) {
   696→            existing.baseIcms += item.baseIcms
   697→            existing.valorIcms += item.valorIcms
   698→            existing.valorIpi += item.valorIpi
   699→            existing.valorPis += item.valorPis
   700→            existing.valorCofins += item.valorCofins
   701→            existing.valorTotal += item.valorTotal
   702→          } else {
   703→            cfopResumos.push({
   704→              cfop,
   705→              cst: cst || '',
   706→              baseIcms: item.baseIcms,
   707→              valorIcms: item.valorIcms,
   708→              valorIpi: item.valorIpi,
   709→              valorPis: item.valorPis,
   710→              valorCofins: item.valorCofins,
   711→              valorTotal: item.valorTotal,
   712→            })
   713→          }
   714→        }
   715→
   716→        itemNum++
   717→      }
   718→    }
   719→
   720→    // |C190| - Resumo de operações por CFOP
   721→    // Group by CFOP
   722→    const cfopMap = new Map<string, CfopResumo>()
   723→    for (const r of cfopResumos) {
   724→      const key = `${r.cfop}_${r.cst}`
   725→      const existing = cfopMap.get(key)
   726→      if (existing) {
   727→        existing.baseIcms += r.baseIcms
   728→        existing.valorIcms += r.valorIcms
   729→        existing.valorIpi += r.valorIpi
   730→        existing.valorPis += r.valorPis
   731→        existing.valorCofins += r.valorCofins
   732→        existing.valorTotal += r.valorTotal
   733→      } else {
   734→        cfopMap.set(key, { ...r })
   735→      }
   736→    }
   737→
   738→    for (const [, resumo] of cfopMap) {
   739→      lines.push({
   740→        registro: 'C190',
   741→        bloco: 'C',
   742→        linha: buildLine([
   743→          'C190',
   744→          resumo.cfop,                     // CFOP
   745→          resumo.cst,                      // CST_ICMS
   746→          formatNumber(resumo.baseIcms),   // VL_BC_ICMS
   747→          formatNumber(resumo.valorIcms),  // VL_ICMS
   748→          '',                              // VL_BC_ICMS_ST
   749→          '',                              // VL_ICMS_ST
   750→          formatNumber(resumo.valorIpi),   // VL_IPI
   751→          formatNumber(resumo.valorPis),   // VL_PIS
   752→          formatNumber(resumo.valorCofins), // VL_COFINS
   753→        ]),
   754→        ordem: ordem++,
   755→      })
   756→    }
   757→  }
   758→
   759→  // |C995| - Encerramento do Bloco C
   760→  const blockCCount = lines.filter(l => l.bloco === 'C').length + 1 // +1 for C995 itself
   761→  lines.push({
   762→    registro: 'C995',
   763→    bloco: 'C',
   764→    linha: buildLine(['C995', String(blockCCount)]),
   765→    ordem: ordem++,
   766→  })
   767→
   768→  return { lines, cfopResumos }
   769→}

   868→async function generateBlocoE(
   869→  config: SpedGeneratorConfig,
   870→  dataInicio: Date,
   871→  dataFim: Date,
   872→  cfopResumos: CfopResumo[]
   873→): Promise<SpedLine[]> {
   874→  const lines: SpedLine[] = []
   875→  let ordem = 0
   876→
   877→  // |E001| - Abertura do Bloco E
   878→  lines.push({
   879→    registro: 'E001',
   880→    bloco: 'E',
   881→    linha: buildLine(['E001', '0']), // Com dados
   882→    ordem: ordem++,
   883→  })
   884→
   885→  // |E100| - Período da apuração do ICMS
   886→  lines.push({
   887→    registro: 'E100',
   888→    bloco: 'E',
   889→    linha: buildLine(['E100', formatDate(dataInicio), formatDate(dataFim)]),
   890→    ordem: ordem++,
   891→  })
   892→
   893→  // |E110| - Apuração do ICMS - Operações próprias
   894→  // Calculate totals from CFOP resumos
   895→  let totalBaseIcms = 0
   896→  let totalValorIcms = 0
   897→  let totalValorIpi = 0
   898→  let totalValorPis = 0
   899→  let totalValorCofins = 0
   900→
   901→  for (const r of cfopResumos) {
   902→    totalBaseIcms += r.baseIcms
   903→    totalValorIcms += r.valorIcms
   904→    totalValorIpi += r.valorIpi
   905→    totalValorPis += r.valorPis
   906→    totalValorCofins += r.valorCofins
   907→  }
   908→
   909→  // Get financial entries for ICMS debits/credits
   910→  const entradas = cfopResumos.filter(r => r.cfop.startsWith('1') || r.cfop.startsWith('2') || r.cfop.startsWith('3'))
   911→  const saidas = cfopResumos.filter(r => r.cfop.startsWith('5') || r.cfop.startsWith('6') || r.cfop.startsWith('7'))
   912→
   913→  const vlOdeb = saidas.reduce((acc, r) => acc + r.valorIcms, 0) // Débitos de ICMS (saídas)
   914→  const vlOcre = entradas.reduce((acc, r) => acc + r.valorIcms, 0) // Créditos de ICMS (entradas)
   915→  const vlSLd = vlOdeb - vlOcre // Saldo devedor
   916→  const vlSldAnt = '' // Saldo credor do período anterior
   917→  const vlSldCred = vlOcre > vlOdeb ? (vlOcre - vlOdeb) : 0 // Saldo credor a transportar
   918→
   919→  lines.push({
   920→    registro: 'E110',
   921→    bloco: 'E',
   922→    linha: buildLine([
   923→      'E110',
   924→      formatNumber(vlOdeb),        // VL_ODEB - Total de débitos
   925→      '',                           // VL_DED - Deduções
   926→      '',                           // VL_ICMS_ST_REST - ICMS ST a restituir
   927→      formatNumber(vlOcre),        // VL_CRED - Total de créditos
   928→      '',                           // VL_ICMS_ST_COMPL - ICMS ST complementar
   929→      '',                           // VL_ICMS_ST_REST_COMPL - ICMS ST restituir complementar
   930→      '',                           // VL_ICMS_ST_COMPL_COMPL - ICMS ST complementar complementar
   931→      '',                           // VL_SLD_ANT - Saldo credor anterior
   932→      vlSLd > 0 ? formatNumber(vlSLd) : '', // VL_SLD_APUR - Saldo devedor apurado
   933→      '',                           // VL_DEDUC - Deduções do saldo devedor
   934→      vlSldCred > 0 ? formatNumber(vlSldCred) : '', // VL_ICMS_RECOLHER - ICMS a recolher
   935→      vlSldCred > 0 ? formatNumber(vlSldCred) : '', // VL_SLD_CRED_TRANSPORTAR - Saldo credor a transportar
   936→      '',                           // VL_DEB_ESP - Débitos especiais
   937→    ]),
   938→    ordem: ordem++,
   939→  })
   940→
   941→  // |E200| - Período de apuração do ICMS - Substituição Tributária
   942→  lines.push({
   943→    registro: 'E200',
   944→    bloco: 'E',
   945→    linha: buildLine(['E200', formatDate(dataInicio), formatDate(dataFim)]),
   946→    ordem: ordem++,
   947→  })
   948→
   949→  // |E500| - Período de apuração do IPI
   950→  lines.push({
   951→    registro: 'E500',
   952→    bloco: 'E',
   953→    linha: buildLine(['E500', formatDate(dataInicio), formatDate(dataFim)]),
   954→    ordem: ordem++,
   955→  })
   956→
   957→  // |E510| - Apuração do IPI
   958→  const cfopIpiMap = new Map<string, { cfop: string; cstIpi: string; valorIpi: number; baseCalculo: number }>()
   959→
   960→  for (const r of cfopResumos) {
   961→    if (r.valorIpi > 0) {
   962→      const key = `${r.cfop}_50` // CST IPI 50 = Saída tributada
   963→      const existing = cfopIpiMap.get(key)
   964→      if (existing) {
   965→        existing.valorIpi += r.valorIpi
   966→        existing.baseCalculo += r.baseIcms
   967→      } else {
   968→        cfopIpiMap.set(key, {
   969→          cfop: r.cfop,
   970→          cstIpi: '50',
   971→          valorIpi: r.valorIpi,
   972→          baseCalculo: r.baseIcms,
   973→        })
   974→      }
   975→    }
   976→  }
   977→
   978→  for (const [, ipiResumo] of cfopIpiMap) {
   979→    lines.push({
   980→      registro: 'E510',
   981→      bloco: 'E',
   982→      linha: buildLine([
   983→        'E510',
   984→        ipiResumo.cfop,                    // CFOP
   985→        ipiResumo.cstIpi,                  // CST_IPI
   986→        formatNumber(ipiResumo.valorIpi),  // VL_CONT_IPI - Valor contábil
   987→        formatNumber(ipiResumo.baseCalculo), // VL_BC_IPI - Base de cálculo
   988→        formatNumber(ipiResumo.valorIpi),  // VL_IPI - Valor do IPI
   989→      ]),
   990→      ordem: ordem++,
   991→    })
   992→  }
   993→
   994→  // |E990| - Encerramento do Bloco E
   995→  const blockECount = lines.filter(l => l.bloco === 'E').length + 1
   996→  lines.push({
   997→    registro: 'E990',
   998→    bloco: 'E',
   999→    linha: buildLine(['E990', String(blockECount)]),
  1000→    ordem: ordem++,
  1001→  })
  1002→
  1003→  return lines
  1004→}

  1093→    registro: '1001',
  1094→    bloco: '1',
  1095→    linha: buildLine(['1001', '1']), // Sem dados
  1096→    ordem: ordem++,
  1097→  })
  1098→
  1099→  // |1990| - Encerramento do Bloco 1
  1100→  const block1Count = lines.filter(l => l.bloco === '1').length + 1
  1101→  lines.push({
  1102→    registro: '1990',
  1103→    bloco: '1',
  1104→    linha: buildLine(['1990', String(block1Count)]),
  1105→    ordem: ordem++,
  1106→  })
  1107→
  1108→  return lines
  1109→}

  1114→async function generateBloco9(allLines: SpedLine[]): Promise<SpedLine[]> {

  1118→  // |9001| - Abertura do Bloco 9

  1151→/**
  1152→ * Generate SPED Fiscal (EFD ICMS/IPI) file content
  1153→ *
  1154→ * @param config Configuration including officeId, competencia, and optional parameters
  1155→ * @returns Generation result with file content, line count, and per-block counts
  1156→ */
  1157→export async function generateSpedFiscal(config: SpedGeneratorConfig): Promise<SpedGenerationResult> {
  1158→  const errors: string[] = []
  1159→
  1160→  try {
  1161→    // Validate competencia format
  1162→    if (!/^\d{6}$/.test(config.competencia)) {
  1163→      return {
  1164→        success: false,
  1165→        conteudo: '',
  1166→        totalLinhas: 0,
  1167→        totalRegistros: {},
  1168→        errors: ['Competência inválida. Use o formato AAAAMM (ex: 202401).'],
  1169→      }
  1170→    }
  1171→
  1172→    // Parse competencia to date range
  1173→    const { dataInicio, dataFim } = parseCompetencia(config.competencia)
  1174→    const effectiveDataInicio = config.dataInicio || dataInicio
  1175→    const effectiveDataFim = config.dataFim || dataFim
  1176→
  1177→    // Verify office exists
  1178→    const office = await db.office.findUnique({
  1179→      where: { id: config.officeId },
  1180→    })
  1181→
  1182→    if (!office) {
  1183→      return {
  1184→        success: false,
  1185→        conteudo: '',
  1186→        totalLinhas: 0,
  1187→        totalRegistros: {},
  1188→        errors: ['Escritório não encontrado.'],
  1189→      }
  1190→    }
  1191→
  1192→    // Generate all blocks
  1193→    const bloco0Lines = await generateBloco0(config, effectiveDataInicio, effectiveDataFim)
  1194→    const { lines: blocoCLines, cfopResumos } = await generateBlocoC(config, effectiveDataInicio, effectiveDataFim)
  1195→    const blocoDLines = await generateBlocoD(config, effectiveDataInicio, effectiveDataFim)
  1196→    const blocoELines = await generateBlocoE(config, effectiveDataInicio, effectiveDataFim, cfopResumos)
  1197→    const blocoHLines = await generateBlocoH(config, effectiveDataFim)
  1198→    const bloco1Lines = await generateBloco1()
  1199→
  1200→    // Combine all lines before Block 9 (needed for total count)
  1201→    const allLinesBefore9 = [
  1202→      ...bloco0Lines,
  1203→      ...blocoCLines,
  1204→      ...blocoDLines,
  1205→      ...blocoELines,
  1206→      ...blocoHLines,
  1207→      ...bloco1Lines,
  1208→    ]
  1209→
  1210→    const bloco9Lines = await generateBloco9(allLinesBefore9)
  1211→
  1212→    // Combine all lines
  1213→    const allLines = [
  1214→      ...allLinesBefore9,
  1215→      ...bloco9Lines,
  1216→    ]
  1217→
  1218→    // Build file content
  1219→    const conteudo = allLines.map(l => l.linha).join('\n')
  1220→
  1221→    // Count per block
  1222→    const totalRegistros: Record<string, number> = {}
  1223→    for (const line of allLines) {
  1224→      totalRegistros[line.bloco] = (totalRegistros[line.bloco] || 0) + 1
  1225→    }
  1226→
  1227→    return {
  1228→      success: true,
  1229→      conteudo,
  1230→      totalLinhas: allLines.length,
  1231→      totalRegistros,
  1232→      errors: errors.length > 0 ? errors : undefined,
  1233→    }
  1234→  } catch (error) {
  1235→    const errorMessage = error instanceof Error ? error.message : 'Erro desconhecido ao gerar SPED Fiscal'
  1236→    console.error('Erro na geração do SPED Fiscal:', error)
  1237→    return {
  1238→      success: false,
  1239→      conteudo: '',
  1240→      totalLinhas: 0,
  1241→      totalRegistros: {},
  1242→      errors: [errorMessage],
  1243→    }
  1244→  }
  1245→}
